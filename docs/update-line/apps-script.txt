const TRIP_DATABASE_ID = '1G8WiHGhHtCXHHdpuIJYJ0kt97RmokG831ETA1bTs7ss';
const TRIP_APPROVED_STATUS = '已核准';
const TRIP_PENDING_STATUS = '待審核';
const TRIP_DEPENDENT_KEY_PREFIX = 'DEP-';
const TRIP_R02_MORNING_HEADER = '第二天上午選擇';
const TRIP_R02_MORNING_CHOICES = ['飯店設施', '水蛙頭步道'];
const TRIP_FUSEN_ROUTES = ['R01-AC', 'R01-BD'];
const TRIP_FUSEN_PENDING = '福森號梯次尚未設定，請聯絡旅遊承辦人。';

function doGet() {
  return tripJson_({ success: true, status: 'ok', service: 'employee-trip', build: 'workbook-vehicle-20261113' });
}

function doPost(e) {
  try {
    const raw = e && e.postData && e.postData.contents;
    let body;
    try { body = JSON.parse(raw || ''); }
    catch (ignored) { return tripJson_({ success: false, status: 'invalid_request' }); }
    const secret = PropertiesService.getScriptProperties().getProperty('API_SECRET');
    if (!secret || !body || typeof body !== 'object' || Array.isArray(body) || body.token !== secret) {
      return tripJson_({ success: false, status: 'unauthorized' });
    }
    const userAction = ['lookup', 'bind', 'trip', 'portal', 'roomAdmin'].includes(body.action);
    if ((!userAction && !['claimReminders', 'ackReminder', 'adminReminders'].includes(body.action)) ||
        (userAction && (typeof body.lineUserId !== 'string' || !/^U[0-9a-f]{32}$/.test(body.lineUserId)))) {
      return tripJson_({ success: false, status: 'invalid_request' });
    }
    if ((body.action === 'bind' || (body.action === 'trip' && body.name !== undefined)) &&
        (typeof body.name !== 'string' || !body.name.trim() || body.name.length > 100 || /[\u0000-\u001f\u007f]/.test(body.name))) {
      return tripJson_({ success: false, status: 'invalid_request' });
    }
    if (body.action === 'ackReminder' &&
        (!tripSafeKey_(body.reminderId) || !tripSafeKey_(body.batchId) ||
         typeof body.retryKey !== 'string' || !/^[0-9a-f]{8}(?:-[0-9a-f]{4}){3}-[0-9a-f]{12}$/.test(body.retryKey))) {
      return tripJson_({ success: false, status: 'invalid_request' });
    }
    const ss = SpreadsheetApp.openById(TRIP_DATABASE_ID);
    let result;
    if (body.action === 'portal') result = tripPortal_(ss, body);
    else if (body.action === 'lookup') result = tripLookup_(ss, body.lineUserId);
    else if (body.action === 'bind') result = tripBind_(ss, body.lineUserId, body.name.trim());
    else if (body.action === 'trip') result = tripItinerary_(ss, body.lineUserId, body.name);
    else if (body.action === 'claimReminders') result = tripClaimReminders_(ss);
    else if (body.action === 'adminReminders') result = tripAdminReminders_(ss, body);
    else if (body.action === 'roomAdmin') result = tripRoomAdmin_(ss, body);
    else result = tripAckReminder_(ss, body);
    return tripJson_(result);
  } catch (ignored) {
    // Do not log bodies, names, IDs, employee rows, tokens or exception text.
    return tripJson_({ success: false, status: 'api_error' });
  }
}

function tripJson_(value) {
  return ContentService.createTextOutput(JSON.stringify(value))
    .setMimeType(ContentService.MimeType.JSON);
}

function tripText_(value) {
  return value == null ? '' : String(value).trim();
}

function tripTable_(ss, name, expectedHeaders) {
  const sheet = ss.getSheetByName(name);
  if (!sheet) throw new Error('Missing table');
  const range = sheet.getDataRange();
  const rows = range.getDisplayValues();
  const headers = rows[0] || [];
  expectedHeaders.forEach(function (expected, index) {
    if (expected !== null && tripText_(headers[index]) !== expected) throw new Error('Invalid schema');
  });
  // Native date cells may display only M/D on phones. Preserve displayed IDs
  // and other columns, but read the full stored year for schedule dates.
  const dateColumn = name === '集合提醒' ? headers.indexOf('集合日期') :
    name === '每日行程' ? headers.indexOf('日期') : -1;
  if (dateColumn >= 0) {
    const values = range.getValues();
    rows.forEach(function (row, index) {
      if (!index) return;
      const value = values[index] && values[index][dateColumn];
      if (Object.prototype.toString.call(value) === '[object Date]' && Number.isFinite(value.getTime())) {
        row[dateColumn] = Utilities.formatDate(value, 'Asia/Taipei', 'yyyy-MM-dd');
      }
    });
  }
  return { sheet: sheet, rows: rows.slice(1), headers: headers };
}

function tripEmployees_(ss) {
  return tripTable_(ss, '員工名單', ['員工ID', '姓名', '路線ID', null, '狀態']);
}

function tripEmployeeRouteIds_(value) {
  const text = tripText_(value).normalize('NFKC');
  if (!text) return [];
  const parts = text.split(/[,、+／\/|;；]+/).map(tripText_);
  if (!parts.length || parts.some(function (part) { return !/^[A-Za-z0-9_.-]{1,40}$/.test(part); }) ||
      parts.length > 2 || new Set(parts).size !== parts.length) return null;
  if (parts.some(function (part) { return /^R01-/i.test(part) && !TRIP_FUSEN_ROUTES.includes(part); })) return null;
  if (parts.filter(function (part) { return part === 'R01' || TRIP_FUSEN_ROUTES.includes(part); }).length > 1) return null;
  return parts;
}

function tripFusenPending_(routeIds) {
  return routeIds.includes('R01') ? TRIP_FUSEN_PENDING : '';
}

function tripRouteImage_(table, route, header) {
  const column = table.headers.indexOf(header);
  const url = column >= 0 ? tripText_(route[column]) : '';
  return url && /^https:\/\/[^\s]+$/i.test(url) && url.length <= 2000 ? url : '';
}

function tripReminderApplies_(reminderRoute, participantRoute) {
  const route = tripText_(reminderRoute);
  return route === participantRoute || (route === 'R01' && TRIP_FUSEN_ROUTES.includes(participantRoute));
}

function tripEmployeeRouteValue_(value) {
  const routeIds = tripEmployeeRouteIds_(value);
  return routeIds && routeIds.length ? routeIds.join(',') : '';
}

function tripRouteLabel_(value) {
  const names = { R01: '福森號（梯次待安排）', 'R01-AC': '福森號第1梯次',
    'R01-BD': '福森號第2梯次', R02: '台中南投' };
  const routes = tripEmployeeRouteIds_(value);
  return routes && routes.length ? routes.map(function (id) { return names[id] || '行程'; }).join('、') : '行程';
}

function tripEmployeeId_(value) {
  const id = tripText_(value).normalize('NFKC');
  return /^[A-Za-z0-9][A-Za-z0-9_.-]{0,63}$/.test(id) ? id : '';
}

function tripDependentKey_(name) {
  const normalized = tripRoomNormalizeName_(name);
  return normalized ? TRIP_DEPENDENT_KEY_PREFIX + tripDigest_('dependent|' + normalized) : '';
}

function tripParticipantKey_(row) {
  const employeeId = tripEmployeeId_(row && row[0]);
  if (employeeId) return employeeId;
  const dependentKey = tripText_(row && row[3]);
  return new RegExp('^' + TRIP_DEPENDENT_KEY_PREFIX + '[0-9a-f]{32}$').test(dependentKey) ? dependentKey : '';
}

function tripR02MorningChoice_(value) {
  const choice = tripText_(value).normalize('NFKC').replace(/\s+/g, '');
  if (!choice) return '';
  return TRIP_R02_MORNING_CHOICES.includes(choice) ? choice : null;
}

function tripAttachR02Morning_(result, access, routeIds) {
  if (!routeIds.includes('R02')) return result;
  const choice = tripR02MorningChoice_(access.employee.morningChoice);
  result.r02MorningChoiceStatus = choice ? 'set' : 'missing';
  if (choice) result.r02MorningChoice = choice;
  return result;
}

  return { success: true, status: status };
}

function tripPortal_(ss, body) {
  const topics = ['行程', '桌次', '房號', '旅遊攻略', '車次'];
  if (body.topic !== undefined && !topics.includes(body.topic)) return tripStatus_('invalid_request');
  if (body.topic === undefined) {
    const valid = function (v) { return typeof v === 'string' && v.trim().length > 0 && v.length <= 50 && !/[\u0000-\u001f\u007f]/.test(v); };
    if (!valid(body.name)) return tripStatus_('name_not_found');
    const result = tripBind_(ss, body.lineUserId, body.name.trim());
    if (result.status !== 'approved') return result;
    return { success: true, status: 'approved', reply: '綁定成功！請點選下方選單查詢行程、桌次、房號或旅遊攻略。' };
  }
  const access = tripLookup_(ss, body.lineUserId, true);
  if (access.status !== 'approved') return access;
  if (body.topic === '行程') return tripItinerary_(ss, body.lineUserId);
  const routeIds = access.employee.routeIds || tripEmployeeRouteIds_(access.employee.routeId);
  let reply;
  if (body.topic === '車次') return { success: true, status: 'approved', reply: tripVehicleReply_(ss, access, routeIds) };
  if (body.topic === '房號') reply = '您的房號：' + (tripMessageText_(access.employee.roomNumber, 100) || '尚未安排');
  if (body.topic === '桌次') {
    if (!routeIds || !routeIds.length) return tripStatus_('route_missing');
    const table = tripTable_(ss, '旅遊路線', ['路線ID', '路線名稱', '交通方式', '開始日期', '結束日期']);
    const index = table.headers.indexOf('桌次圖片網址');
    const urls = [];
    for (let routeIndex = 0; routeIndex < routeIds.length; routeIndex++) {
      if (routeIds[routeIndex] === 'R01') continue;
      const routes = table.rows.filter(function (row) { return tripText_(row[0]) === routeIds[routeIndex]; });
      if (routes.length !== 1) return tripStatus_('route_missing');
      const url = index >= 0 ? tripText_(routes[0][index]) : '';
      if (url && /^https:\/\/[^\s]+$/i.test(url) && url.length <= 2000) urls.push(url);
    }
    const diningReply = tripDiningReply_(ss, access, routeIds);
    if (urls.length === 1) {
      const result = { success: true, status: 'approved', imageUrl: urls[0] };
      if (diningReply) result.diningReply = diningReply;
      return result;
    }
    if (urls.length > 1) {
      const result = { success: true, status: 'approved', imageUrls: urls };
      if (diningReply) result.diningReply = diningReply;
      return result;
    }
    reply = diningReply || tripFusenPending_(routeIds) || '桌次圖片尚未公布。';
    if (diningReply) return { success: true, status: 'approved', reply: reply, diningReply: diningReply };
  }
  if (body.topic === '旅遊攻略') {
    if (!routeIds || !routeIds.length) return tripStatus_('route_missing');
    const rows = tripTable_(ss, '旅遊攻略', ['路線ID', '標題', '內容或連結', '公開狀態']).rows.filter(function (row) {
      return tripText_(row[0]) !== 'R01' && routeIds.includes(tripText_(row[0])) && ['', '公開'].includes(tripText_(row[3]));
    });
    reply = rows.length ? '旅遊攻略\n' + rows.map(function (row) {
      const routeLabel = routeIds.length > 1 ? '【' + tripRouteLabel_(tripText_(row[0])) + '】' : '';
      return routeLabel + tripMessageText_(row[1], 150) + '\n' + tripText_(row[2]).slice(0, 2000);
    }).join('\n\n') : '旅遊攻略尚未公布。';
    if (tripFusenPending_(routeIds)) reply = tripFusenPending_(routeIds) + (rows.length ? '\n\n' + reply : '');
  }
  return { success: true, status: 'approved', reply: reply.slice(0, 4500) };
}

function tripLookup_(ss, lineUserId, includeDetails) {
  const bindings = tripBindings_(ss).rows;
  const own = bindings.filter(function (row) { return tripText_(row[0]) === lineUserId; });
  if (own.length === 0) return tripStatus_('unbound');
  if (own.length !== 1) return tripStatus_('bind_rejected');
  const binding = own[0];
  const status = tripText_(binding[3]);
  if (status === TRIP_PENDING_STATUS) return tripStatus_('pending');
  if (status !== TRIP_APPROVED_STATUS) return tripStatus_('disabled');
  const participantKey = tripText_(binding[1]);
  if (!participantKey || bindings.some(function (row) {
    return tripText_(row[1]) === participantKey && tripText_(row[0]) !== lineUserId;
  })) return tripStatus_('bind_rejected');
  const employeeTable = tripEmployees_(ss);
  const matches = employeeTable.rows.filter(function (row) { return tripParticipantKey_(row) === participantKey; });
  if (matches.length !== 1) return tripStatus_('bind_rejected');
  const employee = matches[0];
  if (tripText_(employee[4]) !== '啟用') return tripStatus_('disabled');
  const routeIds = tripEmployeeRouteIds_(employee[2]);
  if (routeIds === null) return tripStatus_('bind_rejected');
  const approvedEmployee = { id: tripEmployeeId_(employee[0]), name: tripText_(employee[1]), routeId: routeIds.join(',') };
  if (routeIds.length > 1) approvedEmployee.routeIds = routeIds;
  if (includeDetails) {
    approvedEmployee.roomNumber = tripText_(employeeTable.headers[5]) === '房間號碼' ? tripText_(employee[5]) : '';
    approvedEmployee.vehicleNumber = tripText_(employeeTable.headers[6]) === '車次' ? tripText_(employee[6]) : '';
    approvedEmployee.diningSeat = tripText_(employeeTable.headers[7]) === '用餐座位' ? tripText_(employee[7]) : '';
    const morningIndex = employeeTable.headers.indexOf(TRIP_R02_MORNING_HEADER);
    approvedEmployee.morningChoice = morningIndex >= 0 ? (tripR02MorningChoice_(employee[morningIndex]) || '') : '';
  }
  return {
    success: true,
    status: 'approved',
    employee: approvedEmployee
  };
}

function tripBind_(ss, lineUserId, name) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(4000)) return tripStatus_('busy');
  try {
    // Read BOTH sides under the same lock: checking only before locking races.
    const employeeTable = tripEmployees_(ss);
    const employees = employeeTable.rows;
    const table = tripBindings_(ss);
    const rows = table.rows;
    const matches = employees.filter(function (row) { return tripText_(row[1]) === name; });
    if (matches.length === 0) return tripStatus_('name_not_found');
    if (matches.length !== 1) return tripStatus_('name_ambiguous');
    const employee = matches[0];
    const participantKey = tripParticipantKey_(employee);
    if (tripText_(employee[4]) !== '啟用') return tripStatus_('disabled');
    // Employee IDs and generated dependent keys are both stable text keys.
    if (!participantKey || employees.filter(function (row) { return tripParticipantKey_(row) === participantKey; }).length !== 1) {
      return tripStatus_('bind_rejected');
    }
    const ownIndices = [];
    rows.forEach(function (row, index) {
      if (tripText_(row[0]) === lineUserId) ownIndices.push(index);
    });
    if (ownIndices.length > 1) return tripStatus_('bind_rejected');
    if (rows.some(function (row) {
      return tripText_(row[1]) === participantKey && tripText_(row[0]) !== lineUserId;
    })) return tripStatus_('employee_already_bound');
    if (ownIndices.length === 1) {
      const index = ownIndices[0];
      const existing = rows[index];
      if (tripText_(existing[1]) !== participantKey) return tripStatus_('bind_rejected');
      const state = tripText_(existing[3]);
      if (state === TRIP_APPROVED_STATUS) return tripStatus_('approved');
      // Never automatically reactivate a revoked/rejected/unknown binding.
      if (state !== TRIP_PENDING_STATUS) return tripStatus_('disabled');
      table.sheet.getRange(index + 2, 4).setValue(TRIP_APPROVED_STATUS);
    } else {
      table.sheet.appendRow([lineUserId, participantKey, new Date(), TRIP_APPROVED_STATUS]);
    }
    SpreadsheetApp.flush();
    // Personal fields are returned only by approved lookup, not by enrollment.
    return tripStatus_('approved');
  } finally {
    lock.releaseLock();
  }
}

// Dates and clock times are interpreted explicitly in Taiwan (UTC+08),
// independently of the Apps Script project's timezone.
function tripTaiwanTime_(dateValue, timeValue) {
  const date = tripText_(dateValue).match(/^(\d{4})[-/](\d{1,2})[-/](\d{1,2})$/);
  const time = tripText_(timeValue).match(/^(\d{1,2}):(\d{2})(?::(\d{2}))?$/);
  if (!date || !time) return null;
  const year = Number(date[1]), month = Number(date[2]), day = Number(date[3]);
  const hour = Number(time[1]), minute = Number(time[2]), second = Number(time[3] || 0);
  if (year < 1900 || month < 1 || month > 12 || day < 1 || day > 31 ||
      hour > 23 || minute > 59 || second > 59) return null;
  const local = new Date(Date.UTC(year, month - 1, day, hour, minute, second));
  if (local.getUTCFullYear() !== year || local.getUTCMonth() !== month - 1 || local.getUTCDate() !== day) return null;
  const pad = function (value) { return String(value).padStart(2, '0'); };
  return {
    ms: local.getTime() - 8 * 60 * 60 * 1000,
    date: year + '-' + pad(month) + '-' + pad(day),
    time: pad(hour) + ':' + pad(minute)
  };
}

function tripReminders_(ss) {
  return tripTable_(ss, '集合提醒', ['提醒ID', '路線ID', '集合日期', '集合時間', '集合地點', '提前分鐘', '發送狀態', '公告ID']);

function tripMessageText_(value, limit) {
  return tripText_(value).replace(/[\u0000-\u001f\u007f]/g, ' ').slice(0, limit);
}

function tripItinerary_(ss, lineUserId, name) {
  if (name !== undefined) {
    const enrollment = tripBind_(ss, lineUserId, name.trim());
    if (enrollment.status !== 'approved') return enrollment;
  }
  const access = tripLookup_(ss, lineUserId, true);
  if (access.status !== 'approved') return access;
  const routeIds = access.employee.routeIds || tripEmployeeRouteIds_(access.employee.routeId);
  if (!routeIds || !routeIds.length) return tripStatus_('route_missing');
  if (routeIds.length === 1 && routeIds[0] === 'R01') {
    return { success: true, status: 'approved', routeAssignmentPending: true, reply: TRIP_FUSEN_PENDING };
  }
  if (routeIds.length > 1) return tripItineraryMultiple_(ss, access, routeIds);
  const routeId = routeIds[0];
  const routeTable = tripTable_(ss, '旅遊路線', ['路線ID', '路線名稱', '交通方式', '開始日期', '結束日期']);
  const routes = routeTable.rows
    .filter(function (row) { return tripText_(row[0]) === routeId; });
  if (routes.length !== 1) return tripStatus_('route_missing');
  const events = tripTable_(ss, '每日行程', ['行程ID', '路線ID', '日期', '開始時間', '活動名稱', '集合地點', '公開狀態', '備註']).rows
    .filter(function (row) { return tripText_(row[1]) === routeId && ['', '公開'].includes(tripText_(row[6])); })
    .map(function (row) { return { row: row, when: tripTaiwanTime_(row[2], row[3]) }; });
  events.sort(function (a, b) {
    return (a.when ? a.when.ms : Number.MAX_SAFE_INTEGER) - (b.when ? b.when.ms : Number.MAX_SAFE_INTEGER);
  });
  let reply = '您的旅遊行程\n路線：' + (tripMessageText_(routes[0][1], 150) || '路線名稱待填') +
    '\n房間號碼：' + (tripMessageText_(access.employee.roomNumber, 100) || '尚未安排') +
    '\n車次：' + (tripMessageText_(access.employee.vehicleNumber, 100) || '尚未安排') +
    '\n用餐座位：' + (tripMessageText_(access.employee.diningSeat, 100) || '尚未安排');
  if (!events.length) reply += '\n\n行程尚未填入。';
  else events.forEach(function (item) {
    const row = item.row;
    const when = item.when ? item.when.date + ' ' + item.when.time : '日期／時間待確認';
    reply += '\n\n' + when + '｜' + (tripMessageText_(row[4], 200) || '活動名稱待填');
    if (tripText_(row[5])) reply += '\n集合地點：' + tripMessageText_(row[5], 200);
    if (tripText_(row[7])) reply += '\n備註：' + tripMessageText_(row[7], 300);
  });
  const now = Date.now();
  const next = tripReminders_(ss).rows
    .filter(function (row) { return tripReminderApplies_(row[1], routeId) && !['取消', '停用'].includes(tripText_(row[6])); })
    .map(function (row) { return { row: row, when: tripTaiwanTime_(row[2], row[3]) }; })
    .filter(function (item) { return item.when && item.when.ms > now && tripText_(item.row[4]); })
    .sort(function (a, b) { return a.when.ms - b.when.ms; })[0];
  if (next) reply += '\n\n下一次集合：' + next.when.date + ' ' + next.when.time +
    '\n集合地點：' + tripMessageText_(next.row[4], 200);
  if (reply.length > 4500) reply = reply.slice(0, 4400) + '\n其餘行程請洽旅遊承辦人。';
  const result = { success: true, status: 'approved', reply: reply };
  const image = tripRouteImage_(routeTable, routes[0], '行程圖片網址');
  if (image) result.imageUrl = image;
  return tripAttachR02Morning_(result, access, routeIds);
}

  const reminderRows = tripReminders_(ss).rows;

    let section = '路線：' + (tripMessageText_(routes[0][1], 150) || routeId);

    else events.forEach(function (item) {

      if (tripText_(row[5])) section += '\n集合地點：' + tripMessageText_(row[5], 200);
      if (tripText_(row[7])) section += '\n備註：' + tripMessageText_(row[7], 300);

      const url = match && match[1];
      if (!routeImage && url && url.length <= 2000 && !imageUrls.includes(url)) imageUrls.push(url);
    });
    const next = reminderRows.filter(function (row) {
      return tripReminderApplies_(row[1], routeId) && !['取消', '停用'].includes(tripText_(row[6]));
    }).map(function (row) { return { row: row, when: tripTaiwanTime_(row[2], row[3]) }; })
      .filter(function (item) { return item.when && item.when.ms > now && tripText_(item.row[4]); })
      .sort(function (a, b) { return a.when.ms - b.when.ms; })[0];
    if (next) section += '\n\n下一次集合：' + next.when.date + ' ' + next.when.time +
      '\n集合地點：' + tripMessageText_(next.row[4], 200);
    sections.push(section);
  }
  let reply = '您的旅遊行程' +
    '\n房間號碼：' + (tripMessageText_(access.employee.roomNumber, 200) || '尚未安排') +
    '\n車次：' + (tripMessageText_(access.employee.vehicleNumber, 200) || '尚未安排') +
    '\n用餐座位：' + (tripMessageText_(access.employee.diningSeat, 200) || '尚未安排') +
    '\n\n' + sections.join('\n\n');
  if (reply.length > 4500) reply = reply.slice(0, 4400) + '\n其餘行程請洽旅遊承辦人。';
  const result = { success: true, status: 'approved', reply: reply };
  if (imageUrls.length) result.imageUrls = imageUrls.slice(0, 5);
  if (tripFusenPending_(routeIds)) {
    result.routeAssignmentPending = true;
    result.pendingRouteReply = TRIP_FUSEN_PENDING;
  }
  return tripAttachR02Morning_(result, access, routeIds);
}

const TRIP_REMINDER_PREFIX = 'TRIP_REMINDER_V1_';
const TRIP_REMINDER_LEASE_MS = 2 * 60 * 1000;
const TRIP_REMINDER_RETENTION_MS = 24 * 60 * 60 * 1000;
const TRIP_REMINDER_MAX_BYTES = 300000;
const TRIP_REMINDER_MAX_PROPERTIES = 120;
const TRIP_REMINDER_MAX_CLAIM_BATCHES = 12;

function tripSafeKey_(value) {
  return typeof value === 'string' && value.length > 0 && value.length <= 200 && !/[\u0000-\u001f\u007f]/.test(value);
}

function tripDigest_(value) {
  return Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, value, Utilities.Charset.UTF_8)
    .map(function (byte) { return (byte & 255).toString(16).padStart(2, '0'); }).join('').slice(0, 32);
}

function tripReminderSpec_(row, index) {
  const when = tripTaiwanTime_(row[2], row[3]);
  const routeId = tripText_(row[1]), place = tripText_(row[4]), id = tripText_(row[0]);
  if (!when || !routeId || !place || (id && !tripSafeKey_(id))) return null;
  const rowNumber = index + 2;
  const hash = id ? tripDigest_(JSON.stringify([id, routeId, when.ms, place])) : null;
  return { rowNumber: rowNumber, hash: hash, reminderId: id,
    routeId: routeId, place: place, when: when, status: tripText_(row[6]) };
}

function tripReminderParse_(cache, key) {
  if (!Object.prototype.hasOwnProperty.call(cache, key)) return null;
  const value = JSON.parse(cache[key]);
  if (!value || typeof value !== 'object' || Array.isArray(value)) throw new Error('Invalid reminder state');
  return value;
}

function tripReminderBytes_(value) {
  return Utilities.newBlob(value, 'application/json').getBytes().length;
}

function tripReminderStoreFits_(cache, updates) {
  const merged = Object.assign({}, cache, updates);
  const keys = Object.keys(merged).filter(function (key) { return key.indexOf(TRIP_REMINDER_PREFIX) === 0; });
  const workKeys = keys.filter(function (key) { return key.indexOf(TRIP_REMINDER_PREFIX + 'R_') !== 0; });
  return workKeys.length <= TRIP_REMINDER_MAX_PROPERTIES && keys.every(function (key) {
    return tripReminderBytes_(merged[key]) <= 8000;
  }) && keys.reduce(function (total, key) {
    return total + tripReminderBytes_(key) + tripReminderBytes_(merged[key]);
  }, 0) <= TRIP_REMINDER_MAX_BYTES;
}

function tripReminderQueue_(cache, updates, key, value) {
  const text = JSON.stringify(value);
  cache[key] = text;
  updates[key] = text;
}

function tripReminderCleanup_(store, cache, now, activeIds) {
  Object.keys(cache).forEach(function (key) {
    if (key.indexOf(TRIP_REMINDER_PREFIX) !== 0) return;
    let value;
    try { value = JSON.parse(cache[key]); } catch (ignored) { return; }
    const isMarker = key.indexOf(TRIP_REMINDER_PREFIX + 'R_') === 0;
    const removeMarker = isMarker && (!value || !value.reminderId || !activeIds[value.reminderId]);
    if (removeMarker || (!isMarker && value && Number.isFinite(value.expiresAt) && value.expiresAt <= now)) {
      store.deleteProperty(key);
      delete cache[key];
    }
  });
}

function tripReminderRecipients_(ss) {
  const employees = tripEmployees_(ss).rows, bindings = tripBindings_(ss).rows;
  const employeeMap = Object.create(null), employeeCounts = Object.create(null);
  const lineCounts = Object.create(null), associationCounts = Object.create(null), result = Object.create(null);
  employees.forEach(function (row) {
    const key = tripParticipantKey_(row);
    if (!key) return;
    employeeCounts[key] = (employeeCounts[key] || 0) + 1;
    employeeMap[key] = row;
  });
  bindings.forEach(function (row) {
    const line = tripText_(row[0]), key = tripText_(row[1]);
    lineCounts[line] = (lineCounts[line] || 0) + 1;
    associationCounts[key] = (associationCounts[key] || 0) + 1;
  });
  bindings.forEach(function (row) {
    const line = tripText_(row[0]), key = tripText_(row[1]), employee = employeeMap[key];
    if (!/^U[0-9a-f]{32}$/.test(line) || tripText_(row[3]) !== TRIP_APPROVED_STATUS ||
        lineCounts[line] !== 1 || associationCounts[key] !== 1 || employeeCounts[key] !== 1 ||
        !employee || tripText_(employee[4]) !== '啟用') return;
    const routes = tripEmployeeRouteIds_(employee[2]);
    if (!routes || !routes.length) return;
    routes.forEach(function (route) {
      if (!result[route]) result[route] = [];
      if (!result[route].includes(line)) result[route].push(line);
      if (TRIP_FUSEN_ROUTES.includes(route)) {
        if (!result.R01) result.R01 = [];
        if (!result.R01.includes(line)) result.R01.push(line);
      }
    });
  });
  Object.keys(result).forEach(function (route) { result[route].sort(); });
  return result;
}

function tripReminderText_(spec, routeNames) {
  return '時間快到囉～僅剩十分鐘，集合地點在' + tripMessageText_(spec.place, 200) + '\n路線：' +
    tripMessageText_(routeNames[spec.routeId] || spec.routeId, 150) +
    '\n集合時間：' + spec.when.date + ' ' + spec.when.time +
    '\n集合地點：' + tripMessageText_(spec.place, 200);
}

function tripReminderComplete_(table, job, cache, updates) {
  const indices = [];
  table.rows.forEach(function (row, index) { if (tripText_(row[0]) === job.reminderId) indices.push(index); });
  const index = indices.length === 1 ? indices[0] : -1;
  const row = index >= 0 && table.rows[index];
  const current = row && tripReminderSpec_(row, index);
  if (current && current.hash === job.hash && tripText_(row[6]) !== '已發送') {
    table.sheet.getRange(index + 2, 7).setValue('已發送');
    SpreadsheetApp.flush();
  }
  job.complete = true;
  tripReminderQueue_(cache, updates, TRIP_REMINDER_PREFIX + 'M_' + job.hash, job);
}

function tripClaimReminders_(ss) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(4000)) return tripStatus_('busy');
  try {
    const now = Date.now(), store = PropertiesService.getScriptProperties(), cache = store.getProperties(), updates = {};
    const table = tripReminders_(ss), batches = [];
    const idCounts = Object.create(null);
    table.rows.forEach(function (row) {
      const id = tripText_(row[0]);
      if (id) idCounts[id] = (idCounts[id] || 0) + 1;
    });
    tripReminderCleanup_(store, cache, now, idCounts);
    let recipients = null, routeNames = null;
    for (let index = 0; index < table.rows.length; index++) {
      if (batches.length >= TRIP_REMINDER_MAX_CLAIM_BATCHES) break;
      let spec = tripReminderSpec_(table.rows[index], index);
      if (!spec || ['取消', '停用'].includes(spec.status) || now < spec.when.ms - 10 * 60 * 1000 || now >= spec.when.ms) continue;
      if (!spec.reminderId) {
        if (spec.status === '已發送') continue;
        const id = 'M-' + Utilities.getUuid();
        if (idCounts[id]) throw new Error('Duplicate generated reminder ID');
        table.sheet.getRange(index + 2, 1).setValue(id);
        SpreadsheetApp.flush();
        table.rows[index][0] = id;
        idCounts[id] = 1;
        spec = tripReminderSpec_(table.rows[index], index);
      }
      if (idCounts[spec.reminderId] !== 1) continue;
      const manifestKey = TRIP_REMINDER_PREFIX + 'M_' + spec.hash;
      const rowKey = TRIP_REMINDER_PREFIX + 'R_' + tripDigest_(spec.reminderId);
      let job = tripReminderParse_(cache, manifestKey);
      const marker = tripReminderParse_(cache, rowKey);
      // Honor explicitly sent rows; editing an already tracked meeting creates
      // a new generation even when its old sent marker is still visible.
      if (spec.status === '已發送' && (!marker || marker.hash === spec.hash)) {
        if (job && !job.complete) { job.complete = true; tripReminderQueue_(cache, updates, manifestKey, job); }
        continue;
      }
      if (job && job.complete) continue;
      if (!job) {
        if (!recipients) recipients = tripReminderRecipients_(ss);
        const to = recipients[spec.routeId] || [];
        if (!to.length) continue; // New bindings may become eligible next minute.
        if (!routeNames) {
          routeNames = Object.create(null);
          const routeCounts = Object.create(null);
          tripTable_(ss, '旅遊路線', ['路線ID', '路線名稱', '交通方式', '開始日期', '結束日期']).rows.forEach(function (row) {
            const id = tripText_(row[0]);
            if (id) { routeCounts[id] = (routeCounts[id] || 0) + 1; routeNames[id] = tripText_(row[1]); }
          });
          Object.keys(routeNames).forEach(function (id) { if (routeCounts[id] !== 1) delete routeNames[id]; });
        }
        if (!Object.prototype.hasOwnProperty.call(routeNames, spec.routeId)) continue;
        const expiresAt = spec.when.ms + TRIP_REMINDER_RETENTION_MS;
        job = { hash: spec.hash, reminderId: spec.reminderId,
          meetingMs: spec.when.ms, expiresAt: expiresAt, batchIds: [], complete: false };
        const fresh = {};
        for (let offset = 0; offset < to.length; offset += 100) {
          const batchId = spec.hash + '-' + (offset / 100);
          job.batchIds.push(batchId);
          fresh[TRIP_REMINDER_PREFIX + 'B_' + batchId] = JSON.stringify({ reminderId: spec.reminderId,
            batchId: batchId, retryKey: Utilities.getUuid(), to: to.slice(offset, offset + 100),
            text: tripReminderText_(spec, routeNames), leaseUntil: 0, acked: false, expiresAt: expiresAt });
        }
        fresh[manifestKey] = JSON.stringify(job);
        fresh[rowKey] = JSON.stringify({ hash: spec.hash, reminderId: spec.reminderId });
        if (!tripReminderStoreFits_(cache, fresh)) continue;
        Object.keys(fresh).forEach(function (key) { cache[key] = fresh[key]; updates[key] = fresh[key]; });
      }
      let allAcked = true;
      for (let batchIndex = 0; batchIndex < job.batchIds.length; batchIndex++) {
        const batchKey = TRIP_REMINDER_PREFIX + 'B_' + job.batchIds[batchIndex];
        const batch = tripReminderParse_(cache, batchKey);
        if (!batch) throw new Error('Missing reminder batch');
        if (batch.acked) continue;
        allAcked = false;
        if (batch.leaseUntil > now || batches.length >= TRIP_REMINDER_MAX_CLAIM_BATCHES) continue;
        batch.leaseUntil = now + TRIP_REMINDER_LEASE_MS;
        tripReminderQueue_(cache, updates, batchKey, batch);
        batches.push({ reminderId: batch.reminderId, batchId: batch.batchId, retryKey: batch.retryKey,
          to: batch.to, text: batch.text });
      }
      if (allAcked) tripReminderComplete_(table, job, cache, updates);
    }
    if (Object.keys(updates).length) store.setProperties(updates, false);
    return { success: true, status: 'ok', batches: batches };
  } finally {
    lock.releaseLock();
  }
}

function tripAckReminder_(ss, request) {
  const match = request.batchId.match(/^([0-9a-f]{32})-\d{1,4}$/);
  if (!match) return tripStatus_('invalid_request');
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(4000)) return tripStatus_('busy');
  try {
    const store = PropertiesService.getScriptProperties(), cache = store.getProperties(), updates = {};
    const manifestKey = TRIP_REMINDER_PREFIX + 'M_' + match[1], batchKey = TRIP_REMINDER_PREFIX + 'B_' + request.batchId;
    const job = tripReminderParse_(cache, manifestKey), batch = tripReminderParse_(cache, batchKey);
    if (!job || !batch || job.expiresAt <= Date.now() || job.reminderId !== request.reminderId ||
        batch.reminderId !== request.reminderId || batch.retryKey !== request.retryKey ||
        !job.batchIds.includes(request.batchId)) return tripStatus_('invalid_request');
    if (!batch.acked) {
      batch.acked = true;
      batch.leaseUntil = 0;
      tripReminderQueue_(cache, updates, batchKey, batch);
    }
    const allAcked = job.batchIds.every(function (id) {
      const saved = tripReminderParse_(cache, TRIP_REMINDER_PREFIX + 'B_' + id);
      return saved && saved.acked;
    });
    if (allAcked && !job.complete) tripReminderComplete_(tripReminders_(ss), job, cache, updates);
    if (Object.keys(updates).length) store.setProperties(updates, false);
    return tripStatus_('ok');
  } finally {
    lock.releaseLock();
  }
}

// Only the authenticated Worker admin endpoint calls this API action.
function tripAdminRevision_(row) {
  return tripDigest_(JSON.stringify(row.slice(0, 8).map(tripText_)));
}

function tripAdminRoutes_(ss) {
  const table = tripTable_(ss, '旅遊路線', ['路線ID', '路線名稱', '交通方式', '開始日期', '結束日期']);
  const recipients = tripReminderRecipients_(ss), counts = Object.create(null);
  table.rows.forEach(function (r) { const id = tripText_(r[0]); if (id) counts[id] = (counts[id] || 0) + 1; });
  return table.rows.filter(function (r) {
    const id = tripText_(r[0]);
    return id && counts[id] === 1 && (!/^R01-/i.test(id) || TRIP_FUSEN_ROUTES.includes(id));
  }).map(function (r) {
    const id = tripText_(r[0]);
    return { id: id, name: tripText_(r[1]) || id, recipientCount: (recipients[id] || []).length };
  });
}

function tripAdminSnapshot_(ss) {
  const routes = tripAdminRoutes_(ss), names = Object.create(null), now = Date.now();
  routes.forEach(function (r) { names[r.id] = r.name; });
  const table = tripReminders_(ss), ids = Object.create(null);
  table.rows.forEach(function (r) { const id = tripText_(r[0]); if (id) ids[id] = (ids[id] || 0) + 1; });
  const reminders = [];
  table.rows.forEach(function (r, index) {
    if (!r.slice(0, 8).some(function (v) { return tripText_(v) !== ''; })) return;
    const when = tripTaiwanTime_(r[2], r[3]), id = tripText_(r[0]), state = tripText_(r[6]);
    const place = tripText_(r[4]), lines = place.split(/\r?\n/), last = lines[lines.length - 1];
    const map = /^https:\/\/[^\s]+$/i.test(last) ? lines.pop() : '';
    const editable = Boolean(when && when.ms > now + 10 * 60000 && ['', '待發送'].includes(state) && (!id || ids[id] === 1));
    reminders.push({ id: id, rowNumber: index + 2, version: tripAdminRevision_(r), routeId: tripText_(r[1]),
      routeName: names[tripText_(r[1])] || tripText_(r[1]), date: when ? when.date : tripText_(r[2]),
      time: when ? when.time : tripText_(r[3]), meetingMs: when ? when.ms : null,
      sendMs: when ? when.ms - 10 * 60000 : null, placeName: lines.join(' '), mapUrl: map,
      status: state, canEdit: editable, canCancel: editable });
  });
  reminders.sort(function (a, b) {
    const aFuture = a.meetingMs && a.meetingMs > now && !['取消', '停用', '已發送'].includes(a.status);
    const bFuture = b.meetingMs && b.meetingMs > now && !['取消', '停用', '已發送'].includes(b.status);
    if (Boolean(aFuture) !== Boolean(bFuture)) return aFuture ? -1 : 1;
    return aFuture ? a.meetingMs - b.meetingMs : (b.meetingMs || 0) - (a.meetingMs || 0);
  });
  return { success: true, status: 'ok', timezone: 'Asia/Taipei', serverNow: now,
    routes: routes, reminders: reminders.slice(0, 100) };
}

function tripAdminPayload_(request) {
  if (!Array.isArray(request.routeIds) || request.routeIds.length < 1 || request.routeIds.length > 4 ||
      request.routeIds.some(function (id) { return typeof id !== 'string' || !/^[A-Za-z0-9_.-]{1,40}$/.test(id); }) ||
      new Set(request.routeIds).size !== request.routeIds.length) return { error: 'invalid_route' };
  if (request.routeIds.includes('R01') && request.routeIds.some(function (id) { return TRIP_FUSEN_ROUTES.includes(id); })) {
    return { error: 'invalid_route' };
  }
  if (typeof request.date !== 'string' || !/^\d{4}-\d{2}-\d{2}$/.test(request.date) ||
      typeof request.time !== 'string' || !/^\d{2}:\d{2}$/.test(request.time)) return { error: 'invalid_time' };
  const when = tripTaiwanTime_(request.date, request.time);
  if (!when) return { error: 'invalid_time' };
  if ((request.placeName !== undefined && typeof request.placeName !== 'string') ||
      (request.mapUrl !== undefined && typeof request.mapUrl !== 'string')) return { error: 'invalid_place' };
  const name = (request.placeName || '').trim().replace(/\s+/g, ' '), map = (request.mapUrl || '').trim();
  if (name.length > 80 || /^[=+\-@]/.test(name) || (map && !/^https:\/\/[^\s/@]+(?:[/?#][^\s]*)?$/i.test(map))) {
    return { error: 'invalid_place' };
  }
  const place = [name, map].filter(Boolean).join('\n');
  if (!place || place.length > 200 || /[\u0000-\u0009\u000b-\u001f\u007f]/.test(place)) return { error: 'invalid_place' };
  return { when: when, place: place, routes: request.routeIds };
}

function tripAdminReminders_(ss, request) {
  const operation = request.operation;
  if (operation === 'list') return tripAdminSnapshot_(ss);
  if (!['create', 'update', 'cancel'].includes(operation)) return tripStatus_('invalid_request');
  let payload;
  if (operation !== 'cancel') {
    payload = tripAdminPayload_(request);
    if (payload.error) return tripStatus_(payload.error);
  }
  if (operation === 'create' && (typeof request.submissionId !== 'string' ||
      !/^[0-9a-f]{8}(?:-[0-9a-f]{4}){3}-[0-9a-f]{12}$/.test(request.submissionId))) return tripStatus_('invalid_request');
  if (operation !== 'create' && (!Number.isInteger(request.rowNumber) || request.rowNumber < 2 ||
      typeof request.id !== 'string' || (request.id && !tripSafeKey_(request.id)) ||
      typeof request.version !== 'string' || !/^[0-9a-f]{32}$/.test(request.version))) return tripStatus_('invalid_request');
  if (operation === 'update' && payload.routes.length !== 1) return tripStatus_('invalid_route');
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(4000)) return tripStatus_('busy');
  try {
    const table = tripReminders_(ss), now = Date.now();
    if (payload) {
      const routes = tripAdminRoutes_(ss).map(function (r) { return r.id; });
      if (!payload.routes.every(function (id) { return routes.includes(id); })) return tripStatus_('invalid_route');
    }
    if (operation === 'create') {
      const pending = [], saved = [];
      for (let i = 0; i < payload.routes.length; i++) {
        const route = payload.routes[i], id = 'ADMIN-' + request.submissionId + '-' + route;
        const matches = table.rows.filter(function (r) { return tripText_(r[0]) === id; });
        if (matches.length > 1) return tripStatus_('conflict');
        if (matches.length === 1) {
          const r = matches[0], when = tripTaiwanTime_(r[2], r[3]);
          if (tripText_(r[1]) !== route || !when || when.ms !== payload.when.ms || tripText_(r[4]) !== payload.place ||
              ['取消', '停用'].includes(tripText_(r[6]))) return tripStatus_('conflict');
          saved.push(id); continue;
        }
        const duplicate = table.rows.some(function (r) {
          const when = tripTaiwanTime_(r[2], r[3]);
          return tripText_(r[1]) === route && when && when.ms === payload.when.ms && !['取消', '停用'].includes(tripText_(r[6]));
        });
        if (duplicate) return tripStatus_('duplicate');
        pending.push([id, route, payload.when.date, payload.when.time, payload.place, '10', '待發送', '']);
        saved.push(id);
      }
      if (!pending.length) return { success: true, status: 'ok', saved: saved };
      if (payload.when.ms < now + 11 * 60000) return tripStatus_('too_soon');
      table.sheet.getRange(table.sheet.getLastRow() + 1, 1, pending.length, 8).setValues(pending);
      SpreadsheetApp.flush();
      return { success: true, status: 'ok', saved: saved };
    }
    let index;
    if (request.id) {
      const indices = [];
      table.rows.forEach(function (r, i) { if (tripText_(r[0]) === request.id) indices.push(i); });
      if (indices.length !== 1) return tripStatus_(indices.length ? 'conflict' : 'not_found');
      index = indices[0];
    } else index = request.rowNumber - 2;
    const row = table.rows[index];
    if (!row) return tripStatus_('not_found');
    if (tripAdminRevision_(row) !== request.version) return tripStatus_('conflict');
    const state = tripText_(row[6]), when = tripTaiwanTime_(row[2], row[3]);
    if (!when || when.ms <= now + 10 * 60000 || !['', '待發送'].includes(state)) return tripStatus_('locked');
    if (operation === 'cancel') {
      table.sheet.getRange(index + 2, 7).setValue('取消');
      SpreadsheetApp.flush();
      return tripStatus_('ok');
    }
    if (payload.when.ms < now + 11 * 60000) return tripStatus_('too_soon');
    if (table.rows.some(function (r, i) {
      const other = tripTaiwanTime_(r[2], r[3]);
      return i !== index && tripText_(r[1]) === payload.routes[0] && other && other.ms === payload.when.ms &&
        !['取消', '停用'].includes(tripText_(r[6]));
    })) return tripStatus_('duplicate');
    const updated = [request.id || 'ADMIN-' + Utilities.getUuid(), payload.routes[0], payload.when.date,
      payload.when.time, payload.place, '10', '待發送', tripText_(row[7])];
    table.sheet.getRange(index + 2, 1, 1, 8).setValues([updated]);
    SpreadsheetApp.flush();
    return tripStatus_('ok');
  } finally { lock.releaseLock(); }
}

function tripRoomValue_(value) {
  const text = tripText_(value).normalize('NFKC').replace(/\s+/g, ' ');
  if (!text || text.length > 40 || /^[=+\-@]/.test(text) || /[\u0000-\u001f\u007f]/.test(text)) return '';
  return text;
}

function tripRoomAdminAccess_(ss, lineUserId) {
  const access = tripLookup_(ss, lineUserId);
  if (access.status !== 'approved' || !access.employee || !access.employee.id) return null;
  const table = tripEmployees_(ss);
  if (tripText_(table.headers[9]) !== '管理權限') return null;
  const rows = [];
  table.rows.forEach(function (row, index) {
    if (tripText_(row[0]) === access.employee.id) rows.push({ row: row, rowNumber: index + 2 });
  });
  if (rows.length !== 1 || tripText_(rows[0].row[9]) !== '管理員') return null;
  return { employeeId: access.employee.id, ownerHash: tripDigest_(lineUserId), rowNumber: rows[0].rowNumber };
}

function tripRoomImportKey_(importId) {
  return TRIP_ROOM_IMPORT_PREFIX + importId;
}

function tripRoomImportId_(value) {
  return typeof value === 'string' && /^[0-9a-f]{8}(?:-[0-9a-f]{4}){3}-[0-9a-f]{12}$/.test(value);
}

function tripRoomCacheRead_(importId, ownerHash) {
  if (!tripRoomImportId_(importId)) return null;
  const text = CacheService.getScriptCache().get(tripRoomImportKey_(importId));
  if (!text) return null;
  let saved;
  try { saved = JSON.parse(text); } catch (ignored) { return null; }
  if (!saved || ![1, 2].includes(saved.version) || saved.ownerHash !== ownerHash || !Array.isArray(saved.assignments) ||
      !Number.isFinite(saved.expiresAt) || saved.expiresAt <= Date.now()) return null;
  if (saved.version === 1 && !saved.kind) saved.kind = 'room';
  if (!['room', 'roster', 'dining', 'vehicle'].includes(saved.kind)) return null;
  return saved;
}

function tripRoomCacheSave_(ownerHash, assignments, kind, metadata) {
  const importId = Utilities.getUuid();
  const saved = { version: 2, kind: kind || 'room', ownerHash: ownerHash,
    expiresAt: Date.now() + TRIP_ROOM_IMPORT_TTL_SECONDS * 1000, assignments: assignments };
  if (metadata && typeof metadata === 'object') saved.metadata = metadata;
  const text = JSON.stringify(saved);
  if (Utilities.newBlob(text, 'application/json').getBytes().length > TRIP_ROOM_IMPORT_MAX_CACHE_BYTES) return null;
  CacheService.getScriptCache().put(tripRoomImportKey_(importId), text, TRIP_ROOM_IMPORT_TTL_SECONDS);
  return importId;
}

function tripRoomXmlDecode_(value) {
  return String(value || '').replace(/<!\[CDATA\[([\s\S]*?)\]\]>/g, '$1').replace(
    /&(?:#(\d+)|#x([0-9a-f]+)|amp|lt|gt|quot|apos);/gi,
    function (entity, decimal, hex) {
      if (decimal) return String.fromCodePoint(Number(decimal));
      if (hex) return String.fromCodePoint(parseInt(hex, 16));
      return { '&amp;': '&', '&lt;': '<', '&gt;': '>', '&quot;': '"', '&apos;': "'" }[entity.toLowerCase()] || entity;
    });
}

function tripRoomXmlRuns_(xml) {
  const values = [], expression = /<t(?:\s[^>]*)?>([\s\S]*?)<\/t>/gi;
  let match;
  while ((match = expression.exec(xml)) !== null) values.push(tripRoomXmlDecode_(match[1].replace(/<[^>]*>/g, '')));
  return values.join('');
}

function tripRoomColumnIndex_(reference) {
  const match = String(reference || '').replace(/\$/g, '').match(/^([A-Z]{1,3})\d+$/i);
  if (!match) return -1;
  let value = 0;
  for (let i = 0; i < match[1].length; i++) value = value * 26 + match[1].toUpperCase().charCodeAt(i) - 64;
  return value - 1;
}

function tripRoomCellPoint_(reference) {
  const match = String(reference || '').replace(/\$/g, '').match(/^([A-Z]{1,3})(\d+)$/i);
  if (!match) return null;
  const column = tripRoomColumnIndex_(match[1] + match[2]), row = Number(match[2]) - 1;
  if (column < 0 || column >= TRIP_ROOM_IMPORT_MAX_COLUMNS || row < 0 || row >= TRIP_ROOM_IMPORT_MAX_ROWS) return null;
  return { row: row, column: column };
}

function tripRoomSharedStrings_(xml) {
  if (!xml) return [];
  const strings = [], expression = /<si(?:\s[^>]*)?>([\s\S]*?)<\/si>/gi;
  let match;
  while ((match = expression.exec(xml)) !== null) {
    if (strings.length >= TRIP_ROOM_IMPORT_MAX_CELLS) throw new Error('Workbook too large');
    strings.push(tripRoomXmlRuns_(match[1]).slice(0, 500));
  }
  return strings;
}

function tripRoomXmlAttribute_(attributes, name) {
  const match = String(attributes || '').match(new RegExp('(?:^|\\s)' + name + '\\s*=\\s*(["\\\'])(.*?)\\1', 'i'));
  return match ? tripRoomXmlDecode_(match[2]) : '';
}

function tripRoomXmlElements_(xml, tag) {
  const prefix = '(?:[A-Za-z_][\\w.-]*:)?';
  const expression = new RegExp('<' + prefix + tag + '\\b([^>]*?)(?:\\/\\s*>|>([\\s\\S]*?)<\\/' + prefix + tag + '\\s*>)', 'gi');
  const result = [];
  let match;
  while ((match = expression.exec(String(xml || ''))) !== null) {
    if (result.length >= TRIP_ROOM_IMPORT_MAX_CELLS) throw new Error('Workbook too large');
    result.push({ attributes: match[1] || '', body: match[2] || '' });
  }
  return result;
}

function tripRoomColorRgb_(hex) {
  const text = String(hex || '').replace(/^#/, '');
  if (!/^(?:[0-9a-f]{6}|[0-9a-f]{8})$/i.test(text)) return null;
  const rgb = text.slice(-6);
  return [parseInt(rgb.slice(0, 2), 16), parseInt(rgb.slice(2, 4), 16), parseInt(rgb.slice(4, 6), 16)];
}

function tripRoomColorTint_(rgb, tint) {
  if (!tint) return rgb;
  const channels = rgb.map(function (v) { return v / 255; });
  const maximum = Math.max.apply(null, channels), minimum = Math.min.apply(null, channels);
  const delta = maximum - minimum, lightness = (maximum + minimum) / 2;
  const saturation = delta ? delta / (1 - Math.abs(2 * lightness - 1)) : 0;
  let hue = 0;
  if (delta) {
    if (maximum === channels[0]) hue = ((channels[1] - channels[2]) / delta + 6) % 6;
    else if (maximum === channels[1]) hue = (channels[2] - channels[0]) / delta + 2;
    else hue = (channels[0] - channels[1]) / delta + 4;
  }
  const adjusted = tint < 0 ? lightness * (1 + tint) : lightness * (1 - tint) + tint;
  const chroma = (1 - Math.abs(2 * adjusted - 1)) * saturation;
  const x = chroma * (1 - Math.abs(hue % 2 - 1)), offset = adjusted - chroma / 2;
  const sectors = [[chroma, x, 0], [x, chroma, 0], [0, chroma, x], [0, x, chroma], [x, 0, chroma], [chroma, 0, x]];
  return sectors[Math.floor(hue) % 6].map(function (v) { return Math.max(0, Math.min(255, Math.round((v + offset) * 255))); });
}

function tripRoomThemeColors_(xml) {
  const scheme = tripRoomXmlElements_(xml, 'clrScheme')[0];
  if (!scheme) return [];
  // Spreadsheet theme keys swap the first light/dark pairs from XML order.
  return ['lt1', 'dk1', 'lt2', 'dk2', 'accent1', 'accent2', 'accent3', 'accent4', 'accent5', 'accent6', 'hlink', 'folHlink']
    .map(function (name) {
      const slot = tripRoomXmlElements_(scheme.body, name)[0];
      if (!slot) return null;
      const srgb = tripRoomXmlElements_(slot.body, 'srgbClr')[0];
      const system = tripRoomXmlElements_(slot.body, 'sysClr')[0];
      return tripRoomColorRgb_(srgb ? tripRoomXmlAttribute_(srgb.attributes, 'val') :
        system ? tripRoomXmlAttribute_(system.attributes, 'lastClr') : '');
    });
}

function tripRoomColorEnvironment_(stylesXml, themeXml) {
  const defaultPalette = ('000000 FFFFFF FF0000 00FF00 0000FF FFFF00 FF00FF 00FFFF ' +
    '000000 FFFFFF FF0000 00FF00 0000FF FFFF00 FF00FF 00FFFF ' +
    '800000 008000 000080 808000 800080 008080 C0C0C0 808080 ' +
    '9999FF 993366 FFFFCC CCFFFF 660066 FF8080 0066CC CCCCFF ' +
    '000080 FF00FF FFFF00 00FFFF 800080 800000 008080 0000FF ' +
    '00CCFF CCFFFF CCFFCC FFFF99 99CCFF FF99CC CC99FF FFCC99 ' +
    '3366FF 33CCCC 99CC00 FFCC00 FF9900 FF6600 666699 969696 ' +
    '003366 339966 003300 333300 993300 993366 333399 333333').split(' ').map(tripRoomColorRgb_);
  const indexed = tripRoomXmlElements_(stylesXml, 'indexedColors')[0];
  const palette = indexed ? tripRoomXmlElements_(indexed.body, 'rgbColor').map(function (color) {
    return tripRoomColorRgb_(tripRoomXmlAttribute_(color.attributes, 'rgb'));
  }) : defaultPalette;
  return { theme: tripRoomThemeColors_(themeXml), indexed: palette };
}

function tripRoomColorValue_(attributes, environment) {
  if (/^(?:1|true)$/i.test(tripRoomXmlAttribute_(attributes, 'auto'))) return null;
  const rgb = tripRoomXmlAttribute_(attributes, 'rgb'), indexed = tripRoomXmlAttribute_(attributes, 'indexed');
  const theme = tripRoomXmlAttribute_(attributes, 'theme');
  let color = rgb ? tripRoomColorRgb_(rgb) : null;
  if (!color && /^\d+$/.test(indexed)) color = environment.indexed[Number(indexed)] || null;
  if (!color && /^\d+$/.test(theme)) color = environment.theme[Number(theme)] || null;
  if (!color) return null;
  const rawTint = tripRoomXmlAttribute_(attributes, 'tint'), tint = rawTint ? Number(rawTint) : 0;
  if (!Number.isFinite(tint) || tint < -1 || tint > 1) return null;
  return tripRoomColorTint_(color, tint);
}

function tripRoomColorVegetarian_(attributes, environment) {
  const rgb = tripRoomColorValue_(attributes, environment);
  return !!rgb && rgb[1] >= rgb[0] + 8 && rgb[1] >= rgb[2] + 8 && Math.max.apply(null, rgb) - Math.min.apply(null, rgb) >= 16;
}

function tripRoomWorkbookStyles_(stylesXml, themeXml) {
  const environment = tripRoomColorEnvironment_(stylesXml, themeXml);
  const fontsElement = tripRoomXmlElements_(stylesXml, 'fonts')[0];
  const fillsElement = tripRoomXmlElements_(stylesXml, 'fills')[0];
  const fonts = tripRoomXmlElements_(fontsElement ? fontsElement.body : '', 'font').map(function (font) {
    const color = tripRoomXmlElements_(font.body, 'color')[0];
    return !!color && tripRoomColorVegetarian_(color.attributes, environment);
  });
  const fills = tripRoomXmlElements_(fillsElement ? fillsElement.body : '', 'fill').map(function (fill) {
    const pattern = tripRoomXmlElements_(fill.body, 'patternFill')[0];
    if (!pattern || tripRoomXmlAttribute_(pattern.attributes, 'patternType').toLowerCase() !== 'solid') return false;
    const color = tripRoomXmlElements_(pattern.body, 'fgColor')[0];
    return !!color && tripRoomColorVegetarian_(color.attributes, environment);
  });
  const baseElement = tripRoomXmlElements_(stylesXml, 'cellStyleXfs')[0];
  const bases = tripRoomXmlElements_(baseElement ? baseElement.body : '', 'xf');
  const stylesElement = tripRoomXmlElements_(stylesXml, 'cellXfs')[0];
  const styles = tripRoomXmlElements_(stylesElement ? stylesElement.body : '', 'xf').map(function (style) {
    const baseIndex = tripRoomXmlAttribute_(style.attributes, 'xfId');
    const base = /^\d+$/.test(baseIndex) ? bases[Number(baseIndex)] : null;
    function id(name, apply) {
      const own = tripRoomXmlAttribute_(style.attributes, name);
      const inherit = base && (own === '' || /^(?:0|false)$/i.test(tripRoomXmlAttribute_(style.attributes, apply)));
      const value = inherit ? tripRoomXmlAttribute_(base.attributes, name) : own;
      return /^\d+$/.test(value) ? Number(value) : 0;
    }
    return { fontVegetarian: fonts[id('fontId', 'applyFont')] === true, fillVegetarian: fills[id('fillId', 'applyFill')] === true };
  });
  return { colors: environment, styles: styles };
}

function tripRoomRichTextVegetarian_(xml, environment, baseFont) {
  const runs = tripRoomXmlElements_(xml, 'r');
  if (!runs.length) return !!baseFont;
  return runs.some(function (run) {
    if (!tripRoomXmlRuns_(run.body).trim()) return false;
    const properties = tripRoomXmlElements_(run.body, 'rPr')[0];
    const color = properties && tripRoomXmlElements_(properties.body, 'color')[0];
    return color ? tripRoomColorVegetarian_(color.attributes, environment) : !!baseFont;
  });
}

function tripDiningCellVegetarian_(rows, row, column) {
  return !!(rows && rows.cellStyles && rows.cellStyles[row + ',' + column] && rows.cellStyles[row + ',' + column].vegetarian === true);
}

function tripRoomWorksheetRows_(xml, sharedStrings, workbookStyles) {
  const rows = [], cellStyles = Object.create(null), rowExpression = /<row\b([^>]*)>([\s\S]*?)<\/row>/gi;
  const columnStyles = Object.create(null);
  if (workbookStyles) tripRoomXmlElements_(xml, 'col').forEach(function (column) {
    const min = Number(tripRoomXmlAttribute_(column.attributes, 'min'));
    const max = Number(tripRoomXmlAttribute_(column.attributes, 'max'));
    const style = tripRoomXmlAttribute_(column.attributes, 'style');
    if (!Number.isInteger(min) || !Number.isInteger(max) || min < 1 || max < min || !/^\d+$/.test(style)) return;
    for (let index = min - 1; index < Math.min(max, TRIP_ROOM_IMPORT_MAX_COLUMNS); index++) columnStyles[index] = Number(style);
  });
  let rowMatch, populated = 0;
  while ((rowMatch = rowExpression.exec(xml)) !== null) {
    const rowAttribute = rowMatch[1].match(/\br="(\d+)"/i);
    const rowIndex = rowAttribute ? Number(rowAttribute[1]) - 1 : rows.length;
    if (!Number.isInteger(rowIndex) || rowIndex < 0 || rowIndex >= TRIP_ROOM_IMPORT_MAX_ROWS) throw new Error('Workbook too large');
    if (!rows[rowIndex]) rows[rowIndex] = [];
    const rowStyle = tripRoomXmlAttribute_(rowMatch[1], 's');
    const customRowStyle = /^(?:1|true)$/i.test(tripRoomXmlAttribute_(rowMatch[1], 'customFormat')) && /^\d+$/.test(rowStyle)
      ? Number(rowStyle) : null;
    const cellExpression = /<c\b([^>]*?)(?:\/\s*>|>([\s\S]*?)<\/c>)/gi;
    let cellMatch;
    while ((cellMatch = cellExpression.exec(rowMatch[2])) !== null) {
      const reference = cellMatch[1].match(/\br="([^"]+)"/i);
      if (!reference) continue;
      const column = tripRoomColumnIndex_(reference[1]);
      if (column < 0 || column >= TRIP_ROOM_IMPORT_MAX_COLUMNS) continue;
      if (++populated > TRIP_ROOM_IMPORT_MAX_CELLS) throw new Error('Workbook too large');
      const body = cellMatch[2] || '';
      const type = (cellMatch[1].match(/\bt="([^"]+)"/i) || [])[1] || '';
      const raw = (body.match(/<v(?:\s[^>]*)?>([\s\S]*?)<\/v>/i) || [])[1];
      let value = '', rich = '';
      if (type === 'inlineStr') { value = tripRoomXmlRuns_(body); rich = body; }
      else if (type === 's') {
        const index = Number(tripRoomXmlDecode_(raw));
        value = Number.isInteger(index) && index >= 0 && index < sharedStrings.length ? sharedStrings[index] : '';
        rich = sharedStrings.richText && sharedStrings.richText[index] || '';
      } else if (raw !== undefined) value = tripRoomXmlDecode_(raw.replace(/<[^>]*>/g, ''));
      rows[rowIndex][column] = String(value || '').slice(0, 500);
      if (value && workbookStyles) {
        const styleId = tripRoomXmlAttribute_(cellMatch[1], 's');
        const index = /^\d+$/.test(styleId) ? Number(styleId) : customRowStyle !== null ? customRowStyle : columnStyles[column] || 0;
        const style = workbookStyles.styles[index] || workbookStyles.styles[0] || {};
        if (style.fillVegetarian || tripRoomRichTextVegetarian_(rich, workbookStyles.colors, style.fontVegetarian)) {
          cellStyles[rowIndex + ',' + column] = { vegetarian: true };
        }
      }
    }
  }
  const mergeExpression = /<mergeCell\b[^>]*\bref="([^"]+)"[^>]*\/?\s*>/gi;
  let merge, mergeCount = 0;
  while ((merge = mergeExpression.exec(xml)) !== null) {
    if (++mergeCount > 1000) throw new Error('Workbook too large');
    const parts = merge[1].split(':'), start = tripRoomCellPoint_(parts[0]), end = tripRoomCellPoint_(parts[1] || parts[0]);
    if (!start || !end || end.row < start.row || end.column < start.column ||
        (end.row - start.row + 1) * (end.column - start.column + 1) > 10000) throw new Error('Workbook too large');
    const value = rows[start.row] && rows[start.row][start.column];
    if (value === undefined || value === '') continue;
    for (let row = start.row; row <= end.row; row++) {
      if (!rows[row]) rows[row] = [];
      for (let column = start.column; column <= end.column; column++) {
        if (rows[row][column] === undefined || rows[row][column] === '') {
          rows[row][column] = value;
          if (cellStyles[start.row + ',' + start.column]) cellStyles[row + ',' + column] = { vegetarian: true };
        }
      }
    }
  }
  const result = rows.map(function (row) { return row || []; });
  if (Object.keys(cellStyles).length) result.cellStyles = cellStyles;
  return result;
}

function tripRoomCsvRows_(text) {
  text = String(text || '').replace(/^\ufeff/, '');
  if (!text || text.length > TRIP_ROOM_IMPORT_MAX_UNZIPPED_BYTES || /\u0000/.test(text)) throw new Error('Invalid CSV');
  let comma = 0, tab = 0, semicolon = 0, quoted = false;
  for (let i = 0; i < Math.min(text.length, 2000); i++) {
    const character = text[i];
    if (character === '"') quoted = !quoted;
    else if (!quoted && (character === '\r' || character === '\n')) break;
    else if (!quoted && character === ',') comma++;
    else if (!quoted && character === '\t') tab++;
    else if (!quoted && character === ';') semicolon++;
  }
  const delimiter = tab > comma && tab >= semicolon ? '\t' : semicolon > comma ? ';' : ',';
  const rows = [], row = [];
  let field = '', inQuotes = false, cells = 0;
  function pushField() {
    if (row.length < TRIP_ROOM_IMPORT_MAX_COLUMNS) row.push(field.slice(0, 500));
    field = '';
    if (++cells > TRIP_ROOM_IMPORT_MAX_CELLS) throw new Error('CSV too large');
  }
  function pushRow() {
    pushField();
    rows.push(row.splice(0));
    if (rows.length > TRIP_ROOM_IMPORT_MAX_ROWS) throw new Error('CSV too large');
  }
  for (let index = 0; index < text.length; index++) {
    const character = text[index];
    if (inQuotes) {
      if (character === '"' && text[index + 1] === '"') { field += '"'; index++; }
      else if (character === '"') inQuotes = false;
      else field += character;
    } else if (character === '"' && !field) inQuotes = true;
    else if (character === delimiter) pushField();
    else if (character === '\n') pushRow();
    else if (character !== '\r') field += character;
  }
  if (inQuotes) throw new Error('Invalid CSV');
  if (field || row.length) pushRow();
  return rows;
}

function tripRoomZipUInt16_(bytes, offset) {
  if (offset < 0 || offset + 2 > bytes.length) return null;
  return (bytes[offset] & 255) | ((bytes[offset + 1] & 255) << 8);
}

function tripRoomZipUInt32_(bytes, offset) {
  if (offset < 0 || offset + 4 > bytes.length) return null;
  return ((bytes[offset] & 255) | ((bytes[offset + 1] & 255) << 8) |
    ((bytes[offset + 2] & 255) << 16) | ((bytes[offset + 3] & 255) * 0x1000000)) >>> 0;
}

// Inspect the central directory before Utilities.unzip allocates the
// uncompressed workbook. ZIP64 and encrypted archives are unsupported.
function tripRoomZipSummary_(bytes) {
  let end = -1;
  for (let index = bytes.length - 22; index >= Math.max(0, bytes.length - 65557); index--) {
    if (tripRoomZipUInt32_(bytes, index) === 0x06054b50) { end = index; break; }
  }
  if (end < 0) return { error: 'encrypted_or_invalid_file' };
  const entries = tripRoomZipUInt16_(bytes, end + 10), centralSize = tripRoomZipUInt32_(bytes, end + 12);
  const centralOffset = tripRoomZipUInt32_(bytes, end + 16), commentLength = tripRoomZipUInt16_(bytes, end + 20);
  if (entries === null || centralSize === null || centralOffset === null || commentLength === null ||
      entries === 0xffff || centralSize === 0xffffffff || centralOffset === 0xffffffff ||
      entries < 1 || entries > TRIP_ROOM_IMPORT_MAX_ENTRIES || end + 22 + commentLength !== bytes.length ||
      centralOffset + centralSize > end) return { error: 'encrypted_or_invalid_file' };
  let offset = centralOffset, uncompressed = 0;
  for (let index = 0; index < entries; index++) {
    if (tripRoomZipUInt32_(bytes, offset) !== 0x02014b50) return { error: 'encrypted_or_invalid_file' };
    const flags = tripRoomZipUInt16_(bytes, offset + 8), size = tripRoomZipUInt32_(bytes, offset + 24);
    const nameLength = tripRoomZipUInt16_(bytes, offset + 28), extraLength = tripRoomZipUInt16_(bytes, offset + 30);
    const entryComment = tripRoomZipUInt16_(bytes, offset + 32);
    if ([flags, size, nameLength, extraLength, entryComment].some(function (value) { return value === null; }) ||
        (flags & 1) || size === 0xffffffff) return { error: 'encrypted_or_invalid_file' };
    uncompressed += size;
    if (uncompressed > TRIP_ROOM_IMPORT_MAX_UNZIPPED_BYTES) return { error: 'file_too_large' };
    offset += 46 + nameLength + extraLength + entryComment;
    if (offset > end) return { error: 'encrypted_or_invalid_file' };
  }
  if (offset !== centralOffset + centralSize) return { error: 'encrypted_or_invalid_file' };
  return { entries: entries, uncompressedBytes: uncompressed };
}

function tripRoomWorkbookTables_(filename, bytes) {
  const lower = filename.toLowerCase();
  if (/\.xls$/.test(lower)) return { error: 'unsupported_format' };
  if (!/\.(xlsx|csv)$/.test(lower)) return { error: 'unsupported_format' };
  if (!bytes.length || bytes.length > TRIP_ROOM_IMPORT_MAX_FILE_BYTES) return { error: 'file_too_large' };
  if (/\.csv$/.test(lower)) {
    const csv = Utilities.newBlob(bytes, 'text/csv', filename).getDataAsString('UTF-8');
    return { tables: [tripRoomCsvRows_(csv)] };
  }
  const zip = tripRoomZipSummary_(bytes);
  if (zip.error) return zip;
  let blobs;
  // XLSX is a ZIP container. Give the unzip service a ZIP Blob explicitly.
  try { blobs = Utilities.unzip(Utilities.newBlob(bytes, 'application/zip', 'workbook.zip')); }
  catch (ignored) { return { error: 'encrypted_or_invalid_file' }; }
  if (!Array.isArray(blobs) || !blobs.length || blobs.length > TRIP_ROOM_IMPORT_MAX_ENTRIES) {
    return { error: 'encrypted_or_invalid_file' };
  }
  const entries = Object.create(null);
  let totalBytes = 0;
  for (let i = 0; i < blobs.length; i++) {
    const name = String(blobs[i].getName() || '').replace(/\\/g, '/').replace(/^\/+/, '');
    const entryBytes = blobs[i].getBytes();
    totalBytes += entryBytes.length;
    if (totalBytes > TRIP_ROOM_IMPORT_MAX_UNZIPPED_BYTES) return { error: 'file_too_large' };
    if (/^(EncryptionInfo|EncryptedPackage)$/i.test(name)) return { error: 'encrypted_or_invalid_file' };
    if (/^(xl\/sharedStrings\.xml|xl\/styles\.xml|xl\/theme\/theme1\.xml|xl\/worksheets\/[^/]+\.xml)$/i.test(name)) {
      entries[name.toLowerCase()] = blobs[i].getDataAsString('UTF-8');
    }
  }
  const shared = tripRoomSharedStrings_(entries['xl/sharedstrings.xml'] || '');
  shared.richText = tripRoomXmlElements_(entries['xl/sharedstrings.xml'] || '', 'si').map(function (string) { return string.body; });
  const styles = tripRoomWorkbookStyles_(entries['xl/styles.xml'] || '', entries['xl/theme/theme1.xml'] || '');
  const sheetNames = Object.keys(entries).filter(function (name) {
    return /^xl\/worksheets\/[^/]+\.xml$/i.test(name);
  }).sort();
  if (sheetNames.length > TRIP_ROOM_IMPORT_MAX_SHEETS) return { error: 'file_too_large' };
  if (!sheetNames.length) return { error: 'encrypted_or_invalid_file' };
  return { tables: sheetNames.map(function (name) { return tripRoomWorksheetRows_(entries[name], shared, styles); }) };
}

function tripRoomHeaderKey_(value) {
  return tripText_(value).normalize('NFKC').toLowerCase().replace(/[\s_\-.:：／\/()（）#]/g, '');
}

function tripRoomHeaderType_(value) {
  const key = tripRoomHeaderKey_(value);
  if (['員工id', '員工編號', '員編', '工號', 'employeeid', 'staffid'].includes(key)) return 'id';
  if (['房號', '房間號碼', '客房號碼', 'room', 'roomno', 'roomnumber'].includes(key)) return 'room';
  if (['姓名', '員工姓名', '中文姓名', 'name', 'guestname'].includes(key) ||
      /^(旅客姓名|房客姓名|入住人|同住人|旅客|房客)\d*$/.test(key)) return 'name';
  return '';
}

// Registration rosters use the travel agency's existing workbook layout.
// Employees and dependants are both participants. Dependants keep A/I blank
// and receive an internal key in D so they can bind by their own name.
function tripRosterHeaderType_(value) {
  const key = tripRoomHeaderKey_(value);
  const roomType = tripRoomHeaderType_(value);
  if (roomType === 'id' || roomType === 'name' || roomType === 'room') return roomType;
  if (['部門', '單位', 'department', 'dept'].includes(key)) return 'department';
  if (['參加aorb', '參加方案', '參加行程', '方案', '路線', '路線id', 'route', 'routeid'].includes(key)) return 'route';
  if (['不參加', '未參加', '不報名'].includes(key)) return 'notParticipating';
  if (['眷屬', '家屬', 'dependent'].includes(key)) return 'dependent';
  if (['梯次', '福森號梯次', '福勝號梯次', '福森梯次', '福森號組合', '福森組合', '福森號行程組合', '福森號分組', '福森號路線', 'fusencombination', 'fusencombo'].includes(key)) return 'fusenCombo';
  if (['狀態', 'status'].includes(key)) return 'status';
  if (['車次', '班次', 'vehicle', 'coach'].includes(key)) return 'vehicle';
  if (['用餐座位', '用餐桌次', '桌次', '餐桌', 'diningseat', 'diningtable', 'mealseat'].includes(key)) return 'dining';
  if (['第二天上午選擇', '第二天上午行程', '第二天1000前選擇', '第二天10點前選擇',
      '1000前選擇', '10點前選擇', '上午行程選擇', '晨間選擇'].includes(key)) return 'morningChoice';
  return '';
}

function tripRosterNamedRoute_(key) {
  if (['台中南投', '南投台中', '台中南投三天兩夜'].includes(key)) return 'R02';
  const match = key.match(/^福(?:森|勝)號(?:(AC|BD|A|B))?(?:第([一二12])梯次|梯次([12]))?$/);
  if (!match) return '';
  const code = match[1] ? (['A', 'AC'].includes(match[1]) ? 'AC' : 'BD') : '';
  const batchNumber = match[2] || match[3];
  const batch = batchNumber ? (['一', '1'].includes(batchNumber) ? 'AC' : 'BD') : '';
  if (code && batch && code !== batch) return '';
  return code || batch ? 'R01-' + (code || batch) : 'R01';
}

function tripRosterRoute_(value) {
  const text = tripText_(value).normalize('NFKC').toUpperCase();
  if (!text) return '';
  const rawParts = text.split(/[+、,，;；|／\/]+/).map(function (part) {
    return part.replace(/[\s_.:：()（）]/g, '');
  }).filter(Boolean);
  const routes = [];
  for (let index = 0; index < rawParts.length; index++) {
    const key = rawParts[index];
    let route = '';
    if (['A', 'A案', '方案A', 'R01'].includes(key)) route = 'R01';
    if (['B', 'B案', '方案B', 'R02'].includes(key)) route = 'R02';
    const combo = key.replace(/^R01-?/, '');
    if (['AC', 'BD'].includes(combo)) route = 'R01-' + combo;
    if (!route) route = tripRosterNamedRoute_(key);
    if (!route) return '';
    if (!routes.includes(route)) routes.push(route);
  }
  if (routes.filter(function (route) { return route === 'R01' || TRIP_FUSEN_ROUTES.includes(route); }).length > 1) return '';
  return ['R01'].concat(TRIP_FUSEN_ROUTES, ['R02']).filter(function (route) { return routes.includes(route); }).join(',');
}

function tripRosterFusenCombo_(value) {
  const text = tripText_(value).normalize('NFKC').toUpperCase().replace(/[\s_.:：()（）]/g, '').replace(/^R01-?/, '');
  if (!text) return '';
  if (['1', '梯次1', '第1梯次', '第一梯次'].includes(text)) return 'AC';
  if (['2', '梯次2', '第2梯次', '第二梯次'].includes(text)) return 'BD';
  if (['AC', 'BD'].includes(text)) return text;
  const route = tripRosterNamedRoute_(text);
  return TRIP_FUSEN_ROUTES.includes(route) ? route.slice(4) : null;
}

function tripRosterResolvedRoutes_(route, existingRoute) {
  const routes = tripEmployeeRouteIds_(route), existing = tripEmployeeRouteIds_(existingRoute) || [];
  const assigned = existing.filter(function (id) { return TRIP_FUSEN_ROUTES.includes(id); });
  if (routes && routes.includes('R01') && assigned.length === 1) {
    return routes.map(function (id) { return id === 'R01' ? assigned[0] : id; }).join(',');
  }
  return route;
}

function tripRosterTruthy_(value) {
  const key = tripText_(value).normalize('NFKC').toLowerCase().replace(/\s+/g, '');
  return ['1', 'true', 'yes', 'y', '是', '有', 'v', '✓', '✔'].includes(key);
}

function tripRosterMergeValue_(items, field) {
  const values = [];
  items.forEach(function (item) {
    const value = tripText_(item[field]);
    if (value && !values.includes(value)) values.push(value);
  });
  return values.length <= 1 ? { value: values[0] || '' } : { conflict: true };
}

function tripRosterMergeCandidates_(candidates, issues) {
  const groups = Object.create(null), merged = [], outputIssues = (issues || []).slice();
  candidates.forEach(function (candidate) {
    const key = candidate.dependent ? 'D:' + tripRoomNormalizeName_(candidate.name) : 'E:' + candidate.employeeId;
    if (!groups[key]) groups[key] = [];
    groups[key].push(candidate);
  });
  Object.keys(groups).forEach(function (key) {
    const items = groups[key], routeIds = [], seenRoutes = Object.create(null);
    if (outputIssues.some(function (issue) { return issue.participantKey && issue.participantKey === items[0].participantKey; })) return;
    let reason = '';
    items.forEach(function (item) {
      const routes = tripEmployeeRouteIds_(item.route) || [];
      routes.forEach(function (route) {
        if (seenRoutes[route]) reason = 'duplicate_file_participant';
        seenRoutes[route] = true;
        if (!routeIds.includes(route)) routeIds.push(route);
      });
    });
    const fields = ['status', 'room', 'vehicle', 'dining', 'department', 'morningChoice', 'fusenCombo'];
    const values = Object.create(null);
    fields.forEach(function (field) {
      const mergedValue = tripRosterMergeValue_(items, field);
      if (mergedValue.conflict) reason = field === 'morningChoice' ? 'conflicting_morning_choice' :
        field === 'fusenCombo' ? 'conflicting_fusen_combo' : 'conflicting_file_data';
      values[field] = mergedValue.value || '';
    });
    const names = [];
    items.forEach(function (item) {
      const name = tripRoomNormalizeName_(item.name);
      if (name && !names.includes(name)) names.push(name);
    });
    if (names.length !== 1) reason = 'conflicting_file_data';
    const fusenRoutes = routeIds.filter(function (id) { return id === 'R01' || TRIP_FUSEN_ROUTES.includes(id); });
    if (fusenRoutes.length > 1) reason = 'conflicting_fusen_combo';
    if (values.fusenCombo && !fusenRoutes.length) reason = 'fusen_combo_not_applicable';
    const assignedRoute = values.fusenCombo ? 'R01-' + values.fusenCombo : '';
    if (assignedRoute && fusenRoutes.length === 1 && fusenRoutes[0] !== 'R01' && fusenRoutes[0] !== assignedRoute) reason = 'conflicting_fusen_combo';
    const route = ['R01'].concat(TRIP_FUSEN_ROUTES, ['R02']).filter(function (id) { return routeIds.includes(id); })
      .map(function (id) { return id === 'R01' && assignedRoute ? assignedRoute : id; }).join(',');
    if (values.morningChoice && !routeIds.includes('R02')) reason = 'morning_choice_not_applicable';
    if (reason) {
      outputIssues.push({ source: items.map(function (item) { return item.source; }).join('、'),
        value: items[0].name || items[0].employeeId, reason: reason });
      return;
    }
    const first = items[0], dependent = !!first.dependent;
    merged.push({ source: items.map(function (item) { return item.source; }).join('、'),
      employeeId: dependent ? '' : first.employeeId, participantKey: dependent ? tripDependentKey_(first.name) : first.employeeId,
      dependent: dependent, name: first.name, route: route, status: values.status,
      room: values.room, vehicle: values.vehicle, dining: values.dining,
      department: dependent ? '' : values.department, morningChoice: values.morningChoice });
  });
  return { candidates: merged, issues: outputIssues };
}

function tripRosterCandidatesFromTables_(tables) {
  const candidates = [], issues = [], rosterTableIndices = [];
  let headerFound = false, ignoredCount = 0;
  tables.forEach(function (rows, tableIndex) {
    let best = null;
    for (let rowIndex = 0; rowIndex < Math.min(rows.length, 30); rowIndex++) {
      const columns = Object.create(null);
      (rows[rowIndex] || []).slice(0, TRIP_ROOM_IMPORT_MAX_COLUMNS).forEach(function (cell, column) {
        const type = tripRosterHeaderType_(cell);
        if (type && columns[type] === undefined) columns[type] = column;
      });
      if (columns.id !== undefined && columns.name !== undefined && columns.route !== undefined) {
        const score = 20 + ['department', 'notParticipating', 'dependent', 'status', 'room', 'vehicle', 'dining', 'morningChoice', 'fusenCombo']
          .filter(function (type) { return columns[type] !== undefined; }).length - rowIndex / 100;
        if (!best || score > best.score) best = { rowIndex: rowIndex, columns: columns, score: score };
      }
    }
    if (!best) return;
    headerFound = true;
    rosterTableIndices.push(tableIndex);
    for (let rowIndex = best.rowIndex + 1; rowIndex < rows.length; rowIndex++) {
      const row = rows[rowIndex] || [], columns = best.columns;
      const source = '工作表' + (tableIndex + 1) + ' 第' + (rowIndex + 1) + '列';
      const rawId = columns.id === undefined ? '' : row[columns.id];
      const rawName = columns.name === undefined ? '' : row[columns.name];
      const rawRoute = columns.route === undefined ? '' : row[columns.route];
      const employeeId = tripRosterEmployeeId_(rawId), name = tripRoomValue_(rawName);
      if (!employeeId && !name && !tripText_(rawRoute)) continue;
      const dependent = columns.dependent !== undefined && tripRosterTruthy_(row[columns.dependent]);
      if (!tripText_(rawRoute) && columns.notParticipating !== undefined && tripRosterTruthy_(row[columns.notParticipating])) {
        ignoredCount++;
        continue;
      }
      const route = tripRosterRoute_(rawRoute);
      let reason = '';
      if (!dependent && !employeeId) reason = tripText_(rawId) ? 'invalid_employee_id' : 'missing_employee_id';
      else if (!name) reason = 'missing_name';
      else if (!tripText_(rawRoute)) reason = 'missing_route';
      else if (!route) reason = 'invalid_route';
      const status = columns.status === undefined ? '' : tripRosterStatus_(row[columns.status]);
      if (!reason && status === null) reason = 'invalid_status';
      const morningChoice = columns.morningChoice === undefined ? '' : tripRosterMorningChoice_(row[columns.morningChoice]);
      if (!reason && morningChoice === null) reason = 'invalid_morning_choice';
      const fusenCombo = columns.fusenCombo === undefined ? '' : tripRosterFusenCombo_(row[columns.fusenCombo]);
      if (!reason && fusenCombo === null) reason = 'invalid_fusen_combo';
      if (reason) {
        issues.push({ source: source, value: name || employeeId || tripText_(rawRoute), reason: reason,
          participantKey: dependent ? tripDependentKey_(name) : employeeId });
        continue;
      }
      candidates.push({ source: source, employeeId: dependent ? '' : employeeId,
        participantKey: dependent ? tripDependentKey_(name) : employeeId, dependent: dependent,
        name: name, route: route, status: status || '',
        room: columns.room === undefined ? '' : tripRoomValue_(row[columns.room]),
        vehicle: columns.vehicle === undefined ? '' : tripRoomValue_(row[columns.vehicle]),
        dining: columns.dining === undefined ? '' : tripRoomValue_(row[columns.dining]),
        department: dependent || columns.department === undefined ? '' : tripRoomValue_(row[columns.department]),
        morningChoice: morningChoice || '', fusenCombo: fusenCombo || '' });
      if (candidates.length > TRIP_ROOM_IMPORT_MAX_ASSIGNMENTS) throw new Error('Too many assignments');
    }
  });
  const merged = tripRosterMergeCandidates_(candidates, issues);
  return { candidates: merged.candidates, issues: merged.issues, headerFound: headerFound, ignoredCount: ignoredCount,
    rosterTableIndices: rosterTableIndices };
}

function tripRoomSplitNames_(value) {
  const text = tripText_(value).normalize('NFKC');
  if (!text) return [];
  let parts = text.split(/[\r\n、,，;；/]+/).map(tripText_).filter(Boolean);
  if (parts.length === 1 && /\s/.test(parts[0])) {
    const spaces = parts[0].split(/\s+/).map(tripText_).filter(Boolean);
    if (spaces.length > 1 && spaces.every(function (part) { return part.length >= 2 && part.length <= 30; })) parts = spaces;
  }
  return parts.filter(function (part) { return part.length <= 50; });
}

function tripRoomCandidatesFromTables_(tables) {
  const result = [], issues = [];
  let headerFound = false;
  tables.forEach(function (rows, tableIndex) {
    let best = null;
    for (let rowIndex = 0; rowIndex < Math.min(rows.length, 30); rowIndex++) {
      const names = [], ids = [], rooms = [];
      (rows[rowIndex] || []).slice(0, TRIP_ROOM_IMPORT_MAX_COLUMNS).forEach(function (cell, column) {
        const type = tripRoomHeaderType_(cell);
        if (type === 'name') names.push(column);
        if (type === 'id') ids.push(column);
        if (type === 'room') rooms.push(column);
      });
      if (rooms.length && (names.length || ids.length)) {
        const score = names.length * 3 + ids.length * 2 - rowIndex / 100;
        if (!best || score > best.score) best = { rowIndex: rowIndex, names: names, ids: ids, room: rooms[0], score: score };
      }
    }
    if (!best) return;
    headerFound = true;
    let carriedRoom = '';
    for (let rowIndex = best.rowIndex + 1; rowIndex < rows.length; rowIndex++) {
      const row = rows[rowIndex] || [];
      if (!row.some(function (cell) { return tripText_(cell); })) { carriedRoom = ''; continue; }
      const explicitRoom = tripRoomValue_(row[best.room]);
      if (explicitRoom) carriedRoom = explicitRoom;
      const room = explicitRoom || carriedRoom;
      const names = [];
      best.names.forEach(function (column) { Array.prototype.push.apply(names, tripRoomSplitNames_(row[column])); });
      const ids = best.ids.map(function (column) { return tripRoomNormalizeId_(row[column]); });
      if (!names.length && !ids.some(Boolean)) continue;
      const count = Math.max(names.length, ids.length);
      if (!room) {
        issues.push({ source: '工作表' + (tableIndex + 1) + ' 第' + (rowIndex + 1) + '列', value: names.join('、') || ids.join('、'), reason: 'missing_room' });
        continue;
      }
      for (let index = 0; index < count; index++) {
        result.push({ source: '工作表' + (tableIndex + 1) + ' 第' + (rowIndex + 1) + '列',
          employeeId: ids[index] || '', name: names[index] || '', room: room });
        if (result.length > TRIP_ROOM_IMPORT_MAX_ASSIGNMENTS) throw new Error('Too many assignments');
      }
    }
  });
  return { candidates: result, issues: issues, headerFound: headerFound };
}

function tripRoomTextCandidates_(text) {
  if (typeof text !== 'string' || !text.trim() || text.length > 20000 || /[\u0000-\u0008\u000b\u000c\u000e-\u001f\u007f]/.test(text)) {
    return { error: 'invalid_text' };
  }
  text = text.trim();
  if (text.indexOf('房號批次更新') === 0) text = text.slice(6).replace(/^[\s:：]+/, '');
  if (!text) return { error: 'invalid_text' };
  const candidates = [], issues = [], lines = text.split(/\r?\n/);
  if (lines.length > TRIP_ROOM_IMPORT_MAX_ROWS) return { error: 'too_many_rows' };
  lines.forEach(function (line, index) {
    const value = tripText_(line);
    if (!value || value === '房號批次更新' || /^房號\s*(?:姓名|入住人)/.test(value)) return;
    const match = value.match(/^([^\s:：,，\t]{1,40})[\s:：,，\t]+(.+)$/);
    if (!match) { issues.push({ source: '第' + (index + 1) + '列', value: value.slice(0, 80), reason: 'invalid_row' }); return; }
    const room = tripRoomValue_(match[1]), names = tripRoomSplitNames_(match[2]);
    if (!room || !names.length) { issues.push({ source: '第' + (index + 1) + '列', value: value.slice(0, 80), reason: 'invalid_row' }); return; }
    names.forEach(function (name) { candidates.push({ source: '第' + (index + 1) + '列', employeeId: '', name: name, room: room }); });
  });
  if (candidates.length > TRIP_ROOM_IMPORT_MAX_ASSIGNMENTS) return { error: 'too_many_rows' };
  return { candidates: candidates, issues: issues, headerFound: true };
}

function tripRosterSummary_(counts, samples, issues, ignoredCount, noChanges) {

function tripRosterPreview_(ss, ownerHash, parsed, pendingTasks) {
  const employeeData = tripRosterEmployeeData_(ss), table = employeeData.table;
  const tasks = Array.isArray(pendingTasks) ? pendingTasks : pendingTasks && pendingTasks.headerFound ? [{ kind: 'dining', parsed: pendingTasks }] : [];
  const keys = Object.create(null), names = Object.create(null);
  table.rows.forEach(function (row, index) {
    const entry = { row: row, rowNumber: index + 2, formulas: employeeData.formulas[index] || [] };
    const key = tripParticipantKey_(row), name = tripRoomNormalizeName_(row[1]);
    if (key) { if (!keys[key]) keys[key] = []; keys[key].push(entry); }
    if (name) { if (!names[name]) names[name] = []; names[name].push(entry); }
  });
  const fileKeys = Object.create(null), fileNames = Object.create(null);
  parsed.candidates.forEach(function (candidate) {
    const key = tripText_(candidate.participantKey), name = tripRoomNormalizeName_(candidate.name);
    if (!fileKeys[key]) fileKeys[key] = [];
    fileKeys[key].push(candidate);
    if (!fileNames[name]) fileNames[name] = [];
    fileNames[name].push(candidate);
  });
  const issues = (parsed.issues || []).slice(0, 200), assignments = [], samples = [];
  let issueCount = (parsed.issues || []).length, newCount = 0, updatedCount = 0, unchangedCount = 0;
  parsed.candidates.forEach(function (candidate) {
    const key = tripText_(candidate.participantKey), normalizedName = tripRoomNormalizeName_(candidate.name);
    let reason = '';
    if (!key) reason = candidate.dependent ? 'missing_name' : 'missing_employee_id';
    else if ((fileKeys[key] || []).length > 1) reason = 'duplicate_file_id';
    else if ((fileNames[normalizedName] || []).length > 1) reason = 'duplicate_file_name';
    let existing = keys[key] || [];
    if (!existing.length && candidate.dependent && names[normalizedName] && names[normalizedName].length === 1 &&
        !tripEmployeeId_(names[normalizedName][0].row[0])) existing = names[normalizedName];
    if (!reason && existing.length > 1) reason = 'ambiguous_employee';
    if (!reason && existing.length === 1 && tripRoomNormalizeName_(existing[0].row[1]) !== normalizedName) reason = 'identity_mismatch';
    if (!reason && !existing.length && names[normalizedName] && names[normalizedName].length) reason = 'name_exists_with_different_id';
    if (reason) {
      issueCount++;
      if (issues.length < 200) issues.push({ source: candidate.source, value: candidate.name || candidate.employeeId, reason: reason });
      return;
    }
    if (!existing.length) {
      const fields = [candidate.name, candidate.route, candidate.dependent ? key : '', candidate.status || '啟用',
        candidate.room || '', candidate.vehicle || '', candidate.dining || '', candidate.department || '', candidate.morningChoice || ''];
      assignments.push({ mode: 'new', rowNumber: 0, employeeId: candidate.employeeId || '', participantKey: key,
        revision: '', fields: fields });
      newCount++;
      if (samples.length < 8) samples.push('新增 ' + (candidate.employeeId || '眷屬') + ' ' + candidate.name + ' → ' + tripRouteLabel_(candidate.route));
      return;
    }
    const entry = existing[0], row = entry.row;
    const fields = [tripText_(row[1]), tripRosterResolvedRoutes_(candidate.route, row[2]),
      candidate.dependent ? key : tripText_(row[3]), candidate.status || '啟用',
      candidate.room || tripText_(row[5]), candidate.vehicle || tripText_(row[6]),
      candidate.dining || tripText_(row[7]), candidate.dependent ? '' : (candidate.department || tripText_(row[8])),
      candidate.morningChoice || tripText_(row[10])];
    const targetColumns = [1, 2, 3, 4, 5, 6, 7, 8, 10];
    let changed = false, protectedFormula = false;
    targetColumns.forEach(function (column, index) {
      if (tripText_(row[column]) !== tripText_(fields[index])) {
        changed = true;
        if (tripText_(entry.formulas[column])) protectedFormula = true;
      }
    });
    if (protectedFormula) {
      issueCount++;
      if (issues.length < 200) issues.push({ source: candidate.source, value: candidate.name, reason: 'protected_formula' });
      return;
    }
    if (!changed) { unchangedCount++; return; }
    assignments.push({ mode: 'update', rowNumber: entry.rowNumber, employeeId: candidate.employeeId || '', participantKey: key,
      revision: tripRosterEmployeeRevision_(row, entry.formulas), fields: fields });
    updatedCount++;
    if (samples.length < 8) samples.push('更新 ' + (candidate.employeeId || '眷屬') + ' ' + candidate.name + ' → ' + tripRouteLabel_(fields[1]));
  });
  const counts = { new: newCount, updated: updatedCount, unchanged: unchangedCount, conflict: issueCount };
  let summary = tripRosterSummary_(counts, samples, issues, parsed.ignoredCount || 0, !assignments.length);
  if (tasks.length && issueCount > 0) {
    // A rejected roster identity must not reach a later name-only plan and
    // accidentally select an older person with the same name. Mixed files
    // require a clean roster before any preview or auxiliary task is cached.
    return { success: true, status: 'no_rows', kind: 'roster', canConfirm: false,
      summary: ('名單有 ' + issueCount + ' 筆問題，請先修正名單問題後重傳原 Excel。\n本次名單、桌次與車次都沒有寫入；其他分頁尚未寫入。\n\n' +
        tripRosterIssueLines_(issues, 6).join('\n')).slice(0, 1800),
      counts: { new: 0, updated: 0, unchanged: unchangedCount, conflict: issueCount },
      inputCount: parsed.candidates.length, newCount: 0, updatedCount: 0, unchangedCount: unchangedCount,
      issueCount: issueCount, issues: issues };
  }
  if (!assignments.length) return { success: true, status: 'no_rows', kind: 'roster', summary: summary,
    counts: counts, inputCount: parsed.candidates.length, newCount: newCount, updatedCount: updatedCount,
    unchangedCount: unchangedCount, issueCount: issueCount, issues: issues };
  const metadata = { counts: counts, samples: samples, ignoredCount: parsed.ignoredCount || 0 };
  if (tasks.length) {
    metadata.pendingImports = tripImportTasksSave_(ownerHash, tasks, 'pending_roster');
    if (!metadata.pendingImports) return tripStatus_('preview_too_large');
    const diningIndex = tasks.findIndex(function (task) { return task.kind === 'dining'; });
    if (diningIndex >= 0) metadata.pendingDiningId = metadata.pendingImports[diningIndex];
    const descriptions = tasks.map(function (task) { return '另有' + tripImportKindLabel_(task.kind) + '工作表：辨識 ' + task.parsed.candidates.length +
      ' 筆姓名，格式問題 ' + (task.parsed.issues || []).length + ' 筆。'; });
    const labels = tasks.map(function (task) { return tripImportKindLabel_(task.kind); }).join('、');
    summary = summary.slice(0, 1300) + '\n\n' + descriptions.join('\n') +
      '\n確認先寫入名單，接著依序另外確認' + labels + '；取消' + labels + '不會回復已確認的名單。';
  }
  const importId = tripRoomCacheSave_(ownerHash, assignments, 'roster', metadata);
  if (!importId) {
    (metadata.pendingImports || []).forEach(function (id) { CacheService.getScriptCache().remove(tripRoomImportKey_(id)); });
    return tripStatus_('preview_too_large');
  }
  return { success: true, status: 'preview', kind: 'roster', importId: importId,
    expiresInSeconds: TRIP_ROOM_IMPORT_TTL_SECONDS, canConfirm: true, summary: summary, counts: counts, samples: samples,
    inputCount: parsed.candidates.length, newCount: newCount, updatedCount: updatedCount,
    unchangedCount: unchangedCount, issueCount: issueCount, issues: issues };
}

function tripRosterConfirmSaved_(ss, saved, importId) {
  if (!saved.assignments.length || saved.assignments.length > TRIP_ROOM_IMPORT_MAX_ASSIGNMENTS) return tripStatus_('invalid_request');
  const data = tripRosterEmployeeData_(ss), rows = data.table.rows;
  const keys = Object.create(null), names = Object.create(null);
  rows.forEach(function (row, index) {
    const key = tripParticipantKey_(row), name = tripRoomNormalizeName_(row[1]);
    if (key) { if (!keys[key]) keys[key] = []; keys[key].push(index); }
    if (name) { if (!names[name]) names[name] = []; names[name].push(key); }
  });
  const newKeys = Object.create(null), newNames = Object.create(null);
  for (let index = 0; index < saved.assignments.length; index++) {
    const assignment = saved.assignments[index], fields = assignment && assignment.fields;
    if (!assignment || !['new', 'update'].includes(assignment.mode) || !Array.isArray(fields) || fields.length !== 9) {
      return tripStatus_('invalid_request');
    }
    const participantKey = tripText_(assignment.participantKey);
    const employeeId = tripRosterEmployeeId_(assignment.employeeId);
    const dependent = !employeeId && new RegExp('^' + TRIP_DEPENDENT_KEY_PREFIX + '[0-9a-f]{32}$').test(participantKey);
    const routeIds = tripEmployeeRouteIds_(fields[1]);
    const morningChoice = tripR02MorningChoice_(fields[8]);
    if ((!employeeId && !dependent) || (employeeId && participantKey !== employeeId) ||
        (dependent && tripText_(fields[2]) !== participantKey) || !routeIds || !routeIds.length ||
        routeIds.some(function (id) { return !['R01', 'R02'].concat(TRIP_FUSEN_ROUTES).includes(id); }) ||
        !['啟用', '停用'].includes(tripText_(fields[3])) || morningChoice === null ||
        (morningChoice && !routeIds.includes('R02'))) return tripStatus_('invalid_request');
    if (assignment.mode === 'update') {
      const rowIndex = assignment.rowNumber - 2, row = rows[rowIndex];
      const existingKey = row && tripParticipantKey_(row);
      if (!row || (employeeId && tripEmployeeId_(row[0]) !== employeeId) ||
          (dependent && (tripEmployeeId_(row[0]) || (existingKey && existingKey !== participantKey) ||
           tripRoomNormalizeName_(row[1]) !== tripRoomNormalizeName_(fields[0]))) ||
          tripRosterEmployeeRevision_(row, data.formulas[rowIndex] || []) !== assignment.revision) return tripStatus_('stale');
    } else {
      const normalizedName = tripRoomNormalizeName_(fields[0]);
      if ((keys[participantKey] && keys[participantKey].length) ||
          (names[normalizedName] && names[normalizedName].length) || newKeys[participantKey] || newNames[normalizedName]) {
        return tripStatus_('stale');
      }
      newKeys[participantKey] = true;
      newNames[normalizedName] = true;
    }
  }
  const updates = saved.assignments.filter(function (assignment) { return assignment.mode === 'update'; });
  const additions = saved.assignments.filter(function (assignment) { return assignment.mode === 'new'; });
  if (!data.morningHeaderReady) data.table.sheet.getRange(1, 11).setValue(TRIP_R02_MORNING_HEADER);
  if (updates.length) {
    const sortedUpdates = updates.slice().sort(function (a, b) { return a.rowNumber - b.rowNumber; });
    const groups = [];
    sortedUpdates.forEach(function (assignment) {
      const group = groups[groups.length - 1];
      if (!group || assignment.rowNumber !== group[group.length - 1].rowNumber + 1) groups.push([assignment]);
      else group.push(assignment);
    });
    groups.forEach(function (group) {
      const firstRow = group[0].rowNumber, height = group.length;
      const routeRange = data.table.sheet.getRange(firstRow, 3, height, 2);
      const routeValues = routeRange.getValues(), routeFormulas = routeRange.getFormulas();
      const detailRange = data.table.sheet.getRange(firstRow, 5, height, 5);
      const detailValues = detailRange.getValues(), detailFormulas = detailRange.getFormulas();
      const morningRange = data.table.sheet.getRange(firstRow, 11, height, 1);
      const morningValues = morningRange.getValues(), morningFormulas = morningRange.getFormulas();
      group.forEach(function (assignment, offset) {
        const fields = assignment.fields;
        routeValues[offset] = [fields[1], fields[2]];
        detailValues[offset] = [fields[3], fields[4], fields[5], fields[6], fields[7]];
        morningValues[offset][0] = fields[8];
      });
      routeRange.setValues(routeValues.map(function (row, index) {
        return row.map(function (value, column) { return tripText_(routeFormulas[index][column]) || value; });
      }));
      detailRange.setValues(detailValues.map(function (row, rowIndex) {
        return row.map(function (value, column) { return tripText_(detailFormulas[rowIndex][column]) || value; });
      }));
      morningRange.setValues(morningValues.map(function (row, index) {
        return [tripText_(morningFormulas[index][0]) || row[0]];
      }));
    });
  }
  if (additions.length) {
    const appendRows = additions.map(function (assignment) {
      const fields = assignment.fields;
      return [assignment.employeeId, fields[0], fields[1], fields[2], fields[3], fields[4], fields[5], fields[6], fields[7], '', fields[8]];
    });
    data.table.sheet.getRange(rows.length + 2, 1, appendRows.length, 11).setValues(appendRows);
  }
  SpreadsheetApp.flush();
  CacheService.getScriptCache().remove(tripRoomImportKey_(importId));
  const metadata = saved.metadata || {}, counts = metadata.counts ||
    { new: additions.length, updated: updates.length, unchanged: 0, conflict: 0 };
  return { success: true, status: 'ok', kind: 'roster',
    summary: '員工名單已更新：新增 ' + additions.length + ' 人、更新 ' + updates.length + ' 人。',
    counts: counts, newCount: additions.length, updatedCount: updates.length };
}

function tripRosterConfirmWithDining_(ss, saved, importId, ownerHash) {
  return tripImportConfirmWithQueue_(ss, saved, importId, ownerHash);
}

function tripImportKindLabel_(kind) {
  return { roster: '員工名單', dining: '桌次', vehicle: '車次', room: '房號' }[kind] || '資料';
}

function tripImportPendingIds_(saved) {
  const meta = saved.metadata || {};
  return Array.isArray(meta.pendingImports) ? meta.pendingImports : meta.pendingDiningId ? [meta.pendingDiningId] : [];
}

function tripImportTasksSave_(ownerHash, tasks, stage) {
  const ids = [];
  try {
    for (let i = 0; i < tasks.length; i++) {
      const task = tasks[i], id = tripRoomCacheSave_(ownerHash, [], task.kind, { stage: stage, parsed: task.parsed });
      if (!id) throw new Error('Pending import too large');
      ids.push(id);
    }
    return ids;
  } catch (ignored) {
    ids.forEach(function (id) { CacheService.getScriptCache().remove(tripRoomImportKey_(id)); });
    return null;
  }
}

function tripImportQueueRead_(ss, saved, ownerHash) {
  const ids = tripImportPendingIds_(saved), queue = [];
  if (ids.length > 3 || new Set(ids).size !== ids.length) return null;
  const routes = ids.length ? tripDiningRoutes_(ss).map(function (route) { return route.id; }) : [];
  for (let i = 0; i < ids.length; i++) {
    const task = tripRoomCacheRead_(ids[i], ownerHash), meta = task && task.metadata;
    if (!task || !['dining', 'vehicle'].includes(task.kind) || !meta || !['pending_roster', 'next_import'].includes(meta.stage) ||
        !meta.parsed || !meta.parsed.headerFound || !Array.isArray(meta.parsed.candidates)) return null;
    const taskRoutes = task.kind === 'dining' ? meta.parsed.routeId ? [meta.parsed.routeId] : [] : meta.parsed.routeIds || [];
    if (!routes.length || !Array.isArray(taskRoutes) || taskRoutes.some(function (id) { return !routes.includes(id); })) return null;
    queue.push({ id: ids[i], saved: task });
  }
  return queue;
}

function tripImportAttachQueue_(ownerHash, result, queue, history) {
  const active = tripRoomCacheRead_(result.importId, ownerHash);
  if (!active || !['preview', 'choose_route'].includes(result.status)) return false;
  const expiresAt = Date.now() + TRIP_ROOM_IMPORT_TTL_SECONDS * 1000;
  const remaining = [];
  for (let i = 0; i < queue.length; i++) {
    const task = queue[i], meta = task.saved && task.saved.metadata;
    if (!task.saved || task.saved.ownerHash !== ownerHash || !meta || !['pending_roster', 'next_import'].includes(meta.stage)) return false;
    const refreshed = Object.assign({}, task.saved, { expiresAt: expiresAt }), text = JSON.stringify(refreshed);
    if (Utilities.newBlob(text, 'application/json').getBytes().length > TRIP_ROOM_IMPORT_MAX_CACHE_BYTES) return false;
    remaining.push({ id: task.id, saved: refreshed, text: text });
  }
  active.expiresAt = expiresAt;
  const meta = active.metadata || (active.metadata = {});
  meta.pendingImports = queue.map(function (task) { return task.id; });
  delete meta.pendingDiningId;
  const dining = queue.filter(function (task) { return task.saved.kind === 'dining'; })[0];
  if (dining) meta.pendingDiningId = dining.id;
  meta.importHistory = history;
  const text = JSON.stringify(active);
  if (Utilities.newBlob(text, 'application/json').getBytes().length > TRIP_ROOM_IMPORT_MAX_CACHE_BYTES) return false;
  // Renew every still-pending payload together with the new active preview.
  // Otherwise an apparently valid later preview could retain a raw task that
  // expired ten minutes after the original upload rather than this step.
  remaining.forEach(function (task) {
    CacheService.getScriptCache().put(tripRoomImportKey_(task.id), task.text, TRIP_ROOM_IMPORT_TTL_SECONDS);
  });
  CacheService.getScriptCache().put(tripRoomImportKey_(result.importId), text, TRIP_ROOM_IMPORT_TTL_SECONDS);
  return true;
}

function tripImportHistoryResult_(result, history) {
  const lines = history.map(function (entry) { return entry.summary; }).filter(Boolean);
  const roster = history.filter(function (entry) { return entry.kind === 'roster'; })[0];
  if (roster) result.rosterSummary = roster.summary;
  if (lines.length) result.summary = (lines.join('\n\n') + (result.summary ? '\n\n' + result.summary : '')).slice(0, 3800);
  return result;
}

function tripImportAdvance_(ss, ownerHash, queue, completed, priorHistory) {
  const history = (priorHistory || []).slice();
  if (completed && completed.summary) history.push({ kind: completed.kind || '', summary: completed.summary, committed: completed.status === 'ok' });
  for (let i = 0; i < queue.length; i++) {
    const task = queue[i];
    try {
      CacheService.getScriptCache().remove(tripRoomImportKey_(task.id));
      const next = task.saved.kind === 'dining'
        ? tripDiningStart_(ss, ownerHash, task.saved.metadata.parsed, completed && completed.kind === 'roster' && completed.status === 'ok')
        : tripVehicleStart_(ss, ownerHash, task.saved.metadata.parsed);
      if (['preview', 'choose_route'].includes(next.status)) {
        if (!tripImportAttachQueue_(ownerHash, next, queue.slice(i + 1), history)) {
          if (next.importId) CacheService.getScriptCache().remove(tripRoomImportKey_(next.importId));
          throw new Error('Cannot preserve remaining imports');
        }
        return tripImportHistoryResult_(next, history);
      }
      if (next.status === 'no_rows') {
        history.push({ kind: task.saved.kind, summary: next.summary || tripImportKindLabel_(task.saved.kind) + '沒有可寫入的資料。', committed: false });
        if (i === queue.length - 1) {
          const base = Object.assign({}, next, { summary: '' });
          return tripImportHistoryResult_(base, history);
        }
        continue;
      }
      throw new Error('Cannot start next import');
    } catch (ignored) {
      const labels = queue.slice(i).map(function (item) { return tripImportKindLabel_(item.saved.kind); }).join('、');
      queue.slice(i).forEach(function (item) { try { CacheService.getScriptCache().remove(tripRoomImportKey_(item.id)); } catch (ignoredCleanup) {} });
      const committed = history.some(function (entry) { return entry.committed; });
      const lastCommitted = history.filter(function (entry) { return entry.committed; }).pop();
      const result = committed ? completed && completed.status === 'ok' ? Object.assign({}, completed, { summary: '' })
        : { success: true, status: 'ok', kind: lastCommitted.kind, summary: '' }
        : { success: true, status: 'no_rows', kind: task.saved.kind, summary: '' };
      if (!result.kind) result.kind = task.saved.kind;
      tripImportHistoryResult_(result, history);
      result.summary = (result.summary + '\n' + labels + '尚未開始匯入，請重新傳送原 Excel；已確認寫入的資料會保留。').slice(0, 3800);
      return result;
    }
  }
  if (completed) return tripImportHistoryResult_(Object.assign({}, completed, { summary: '' }), history);
  return tripStatus_('no_rows');
}

function tripImportTasksStart_(ss, ownerHash, tasks, prior) {
  const ids = tripImportTasksSave_(ownerHash, tasks, 'next_import');
  if (!ids) return tripStatus_('preview_too_large');
  const saved = { metadata: { pendingImports: ids } }, queue = tripImportQueueRead_(ss, saved, ownerHash);
  if (!queue) return tripStatus_('expired');
  return tripImportAdvance_(ss, ownerHash, queue, prior, []);
}

function tripImportConfirmWithQueue_(ss, saved, importId, ownerHash) {
  if (saved.metadata && ['pending_roster', 'next_import'].includes(saved.metadata.stage)) return tripStatus_('invalid_request');
  const queue = tripImportQueueRead_(ss, saved, ownerHash);
  if (!queue) return tripStatus_('expired');
  const result = saved.kind === 'roster' ? tripRosterConfirmSaved_(ss, saved, importId)
    : saved.kind === 'dining' ? tripDiningConfirmSaved_(ss, saved, importId) : tripVehicleConfirmSaved_(ss, saved, importId);
  if (result.status !== 'ok') return result;
  const history = saved.metadata && saved.metadata.importHistory || [];
  return !queue.length && !history.length ? result : tripImportAdvance_(ss, ownerHash, queue, result, history);
}

function tripRoomIssueLines_(issues, limit) {
  const labels = { missing_room: '缺少房號', invalid_row: '格式錯誤', ambiguous_employee: '重複姓名或編號',
    employee_not_found: '查無', employee_disabled: '停用', room_formula_protected: '原房號為公式',
    conflicting_room: '房號衝突', identity_mismatch: '姓名與員編不一致' };
  return issues.slice(0, limit).map(function (issue) {
    return tripMessageText_(issue.value || issue.source || '未辨識資料', 60) + '：' + (labels[issue.reason] || '未匯入');
  });
}

function tripRoomPreview_(ss, ownerHash, parsed) {
  const employeeData = tripRoomEmployeeData_(ss), table = employeeData.table;
  const ids = Object.create(null), names = Object.create(null);
  table.rows.forEach(function (row, index) {
    const entry = { row: row, rowNumber: index + 2, formula: tripText_((employeeData.formulas[index] || [])[0]) };
    const id = tripRoomNormalizeId_(row[0]), name = tripRoomNormalizeName_(row[1]);
    if (id) { if (!ids[id]) ids[id] = []; ids[id].push(entry); }
    if (name) { if (!names[name]) names[name] = []; names[name].push(entry); }
  });
  const issues = (parsed.issues || []).slice(0, 200), selected = Object.create(null);
  let issueCount = (parsed.issues || []).length;
  parsed.candidates.forEach(function (candidate) {
    const id = tripRoomNormalizeId_(candidate.employeeId), name = tripRoomNormalizeName_(candidate.name);
    let matches = null;
    if (id) matches = ids[id] && ids[id].length === 1 ? ids[id] : null;
    else if (name && names[name] && names[name].length === 1) matches = names[name];
    let reason = '';
    if (!matches) reason = (id && ids[id] && ids[id].length > 1) || (!id && name && names[name] && names[name].length > 1) ? 'ambiguous_employee' : 'employee_not_found';
    else if (name && tripRoomNormalizeName_(matches[0].row[1]) !== name) reason = 'identity_mismatch';
    else if (tripText_(matches[0].row[4]) !== '啟用') reason = 'employee_disabled';
    else if (matches[0].formula) reason = 'room_formula_protected';
    if (reason) {
      issueCount++;
      if (issues.length < 200) issues.push({ source: candidate.source, value: candidate.name || candidate.employeeId, room: candidate.room, reason: reason });
      return;
    }
    const match = matches[0], key = tripParticipantKey_(match.row);
    if (!key) {
      issueCount++;
      if (issues.length < 200) issues.push({ source: candidate.source, value: candidate.name || candidate.employeeId,
        room: candidate.room, reason: 'employee_not_found' });
      return;
    }
    const assignment = { rowNumber: match.rowNumber, participantKey: key,
      employeeId: tripEmployeeId_(match.row[0]), name: tripText_(match.row[1]), room: candidate.room,
      previousRoom: tripText_(match.row[5]), revision: tripRoomEmployeeRevision_(match.row, match.formula) };
    if (!selected[key]) selected[key] = { assignment: assignment, conflict: false };
    else if (selected[key].assignment.room !== assignment.room) {
      selected[key].conflict = true;
      issueCount++;
      if (issues.length < 200) issues.push({ source: candidate.source, value: assignment.name, room: candidate.room, reason: 'conflicting_room' });
    }
  });
  const assignments = Object.keys(selected).filter(function (key) { return !selected[key].conflict; })
    .map(function (key) { return selected[key].assignment; }).sort(function (a, b) { return a.rowNumber - b.rowNumber; });
  if (!assignments.length) {
    const noRowsSamples = tripRoomIssueLines_(issues, 6);
    return { success: true, status: 'no_rows', summary: ('沒有可匯入的房號資料。請確認姓名、員工編號與房號欄位。' +
      (noRowsSamples.length ? '\n' + noRowsSamples.join('\n') : '') +
      (issueCount > noRowsSamples.length ? '\n其餘 ' + (issueCount - noRowsSamples.length) + ' 筆未列出。' : '')).slice(0, 1800),
    inputCount: parsed.candidates.length, matchedCount: 0, issueCount: issueCount, issues: issues };
  }
  const importId = tripRoomCacheSave_(ownerHash, assignments);
  if (!importId) return tripStatus_('preview_too_large');
  const samples = assignments.slice(0, 8).map(function (assignment) { return assignment.name + ' → ' + assignment.room; });
  const issueSamples = tripRoomIssueLines_(issues, 6);
  const issueText = issueCount ? '\n未匯入 ' + issueCount + ' 筆：\n' + issueSamples.join('\n') +
    (issueCount > issueSamples.length ? '\n其餘 ' + (issueCount - issueSamples.length) + ' 筆未列出。' : '') : '';
  const summary = '已辨識 ' + assignments.length + ' 人，待確認寫入。\n' + samples.join('\n') +
    (assignments.length > samples.length ? '\n其餘 ' + (assignments.length - samples.length) + ' 人未列出。' : '') + issueText;
  return { success: true, status: 'preview', importId: importId, expiresInSeconds: TRIP_ROOM_IMPORT_TTL_SECONDS,
    canConfirm: true, summary: summary.slice(0, 1800), inputCount: parsed.candidates.length, matchedCount: assignments.length,
    issueCount: issueCount, matched: assignments.slice(0, 200).map(function (assignment) {
      return { name: assignment.name, room: assignment.room, previousRoom: assignment.previousRoom };
    }), issues: issues };
}

function tripRoomConfirm_(ss, request) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(4000)) return tripStatus_('busy');
  try {
    const access = tripRoomAdminAccess_(ss, request.lineUserId);
    if (!access) return tripStatus_('forbidden');
    const saved = tripRoomCacheRead_(request.importId, access.ownerHash);
    if (!saved) return tripStatus_('expired');
    if (saved.kind === 'roster') return tripRosterConfirmWithDining_(ss, saved, request.importId, access.ownerHash);
    if (['dining', 'vehicle'].includes(saved.kind)) return tripImportConfirmWithQueue_(ss, saved, request.importId, access.ownerHash);
    if (!saved.assignments.length || saved.assignments.length > TRIP_ROOM_IMPORT_MAX_ASSIGNMENTS) return tripStatus_('invalid_request');
    const data = tripRoomEmployeeData_(ss), rows = data.table.rows;
    for (let index = 0; index < saved.assignments.length; index++) {
      const assignment = saved.assignments[index], rowIndex = assignment.rowNumber - 2;
      const row = rows[rowIndex], formula = tripText_((data.formulas[rowIndex] || [])[0]);
      if (!row || tripParticipantKey_(row) !== assignment.participantKey || formula ||
          tripRoomEmployeeRevision_(row, formula) !== assignment.revision) return tripStatus_('stale');
    }
    const range = data.table.sheet.getRange(2, 6, rows.length, 1), values = range.getValues(), formulas = range.getFormulas();
    saved.assignments.forEach(function (assignment) { values[assignment.rowNumber - 2][0] = assignment.room; });
    const output = values.map(function (row, index) {
      const formula = tripText_((formulas[index] || [])[0]);
      return [formula || row[0]];
    });
    range.setValues(output);
    SpreadsheetApp.flush();
    CacheService.getScriptCache().remove(tripRoomImportKey_(request.importId));
    return { success: true, status: 'ok', summary: '已更新 ' + saved.assignments.length + ' 人的房號。',
      updatedCount: saved.assignments.length };
  } finally { lock.releaseLock(); }
}

function tripRoomCancel_(ss, request) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(4000)) return tripStatus_('busy');
  try {
    const access = tripRoomAdminAccess_(ss, request.lineUserId);
    if (!access) return tripStatus_('forbidden');
    const saved = tripRoomCacheRead_(request.importId, access.ownerHash);
    if (!saved) return tripStatus_('expired');
    CacheService.getScriptCache().remove(tripRoomImportKey_(request.importId));
    const pendingIds = tripImportPendingIds_(saved);
    if (pendingIds.length) {
      const labels = [tripImportKindLabel_(saved.kind)];
      pendingIds.forEach(function (id) {
        const task = tripRoomCacheRead_(id, access.ownerHash);
        if (task) labels.push(tripImportKindLabel_(task.kind));
        CacheService.getScriptCache().remove(tripRoomImportKey_(id));
      });
      const suffix = (saved.metadata && saved.metadata.importHistory || []).some(function (entry) { return entry.committed; })
        ? '已確認寫入的資料保留。' : '';
      const last = labels.pop(), label = labels.length ? labels.join('、') + '與' + last : last;
      return { success: true, status: 'ok', kind: saved.kind, summary: '已取消，本次' + label + '資料都沒有寫入。' + suffix };
    }
    if (saved.kind === 'dining') return { success: true, status: 'ok', kind: 'dining',
      summary: '已取消，本次桌次資料沒有寫入。' };
    if (saved.kind === 'vehicle') return { success: true, status: 'ok', kind: 'vehicle',
      summary: '已取消，本次車次資料沒有寫入。' };
    if (saved.kind === 'roster') return { success: true, status: 'ok', kind: 'roster',
      summary: '已取消，本次員工名單沒有寫入。' };
    return { success: true, status: 'ok', summary: '已取消，本次房號資料沒有寫入。' };
  } finally { lock.releaseLock(); }
}

function tripRoomAdmin_(ss, request) {
  const operation = request.operation;
  if (!['authorize', 'previewText', 'previewFile', 'selectDiningRoute', 'confirm', 'cancel'].includes(operation)) return tripStatus_('invalid_request');
  if (operation === 'confirm' || operation === 'cancel') {
    if (!tripRoomImportId_(request.importId)) return tripStatus_('invalid_request');
    return operation === 'confirm' ? tripRoomConfirm_(ss, request) : tripRoomCancel_(ss, request);
  }
  const access = tripRoomAdminAccess_(ss, request.lineUserId);
  if (!access) return tripStatus_('forbidden');
  if (operation === 'authorize') return tripStatus_('approved');
  if (operation === 'selectDiningRoute') return tripDiningSelectRoute_(ss, access, request);
  let parsed;
  if (operation === 'previewText') parsed = tripRoomTextCandidates_(request.text);
  else {
    if (typeof request.fileName !== 'string' || request.fileName.length < 1 || request.fileName.length > 255 ||
        /[\\/\u0000-\u001f\u007f]/.test(request.fileName) || !Number.isInteger(request.fileSize) ||
        request.fileSize < 1 || request.fileSize > TRIP_ROOM_IMPORT_MAX_FILE_BYTES || typeof request.fileBase64 !== 'string' ||
        request.fileBase64.length > Math.ceil(TRIP_ROOM_IMPORT_MAX_FILE_BYTES * 4 / 3) + 8 ||
        !/^[A-Za-z0-9+/]*={0,2}$/.test(request.fileBase64)) return tripStatus_('invalid_request');
    let bytes;
    try { bytes = Utilities.base64Decode(request.fileBase64); }
    catch (ignored) { return tripStatus_('invalid_request'); }
    if (bytes.length !== request.fileSize) return tripStatus_('invalid_request');
    let workbook;
    try { workbook = tripRoomWorkbookTables_(request.fileName, bytes); }
    catch (ignored) { return tripStatus_('invalid_file'); }
    if (workbook.error) return tripStatus_(workbook.error);
    let roster;
    try { roster = tripRosterCandidatesFromTables_(workbook.tables); }
    catch (ignored) { return tripStatus_('file_too_large'); }
    let dining, vehicle;
    try {
      // Keep original worksheet positions for error locations. A roster's
      // optional dining column is not a separate seating-plan worksheet.
      const remainingTables = workbook.tables.map(function (table, index) {
        return (roster.rosterTableIndices || []).includes(index) ? [] : table;
      });
      vehicle = tripVehicleCandidatesFromTables_(remainingTables);
      const diningTables = remainingTables.map(function (table, index) {
        return (vehicle.vehicleTableIndices || vehicle.carTableIndices || []).includes(index) ? [] : table;
      });
      dining = tripDiningCandidatesFromTables_(diningTables);
    }
    catch (ignored) { return tripStatus_('file_too_large'); }
    if (vehicle.error) return tripStatus_(vehicle.error);
    if (dining.error) return tripStatus_(dining.error);
    const tasks = [];
    if (dining.headerFound) tasks.push({ kind: 'dining', parsed: dining });
    if (vehicle.headerFound) tasks.push({ kind: 'vehicle', parsed: vehicle });
    if (roster.headerFound) {
      if (dining.headerFound) {
        const routes = tripDiningRoutes_(ss);
        if (!routes.length || (dining.routeId && !routes.some(function (r) { return r.id === dining.routeId; }))) return tripStatus_('invalid_dining_route');
      }
      if (!roster.candidates.length && !(roster.issues && roster.issues.length)) {
        return tasks.length ? tripImportTasksStart_(ss, access.ownerHash, tasks, null) : tripStatus_('no_rows');
      }
      const result = tripRosterPreview_(ss, access.ownerHash, roster, tasks);
      if (tasks.length && result.status === 'no_rows') {
        if (result.issueCount > 0) return result;
        return tripImportTasksStart_(ss, access.ownerHash, tasks, result);
      }
      return result;
    }
    if (tasks.length) return tripImportTasksStart_(ss, access.ownerHash, tasks, null);
    try { parsed = tripRoomCandidatesFromTables_(workbook.tables); }
    catch (ignored) { return tripStatus_('file_too_large'); }
  }
  if (parsed.error) return tripStatus_(parsed.error);
  if (!parsed.headerFound) return tripStatus_('columns_not_found');
  if (!parsed.candidates.length && !(parsed.issues && parsed.issues.length)) return tripStatus_('no_rows');
  return tripRoomPreview_(ss, access.ownerHash, parsed);
}

// One route keeps one fixed seating plan for the entire trip. The complete
// route plan is replaced only after its owner confirms an unchanged preview.
const TRIP_DINING_HEADERS = ['人員識別碼', '路線ID', '桌號', '座位序號', '姓名', '飲食習慣'];

function tripDiningRoutes_(ss) {
  const table = tripTable_(ss, '旅遊路線', ['路線ID', '路線名稱', '交通方式', '開始日期', '結束日期']);
  const allowed = TRIP_FUSEN_ROUTES.concat(['R02']), counts = Object.create(null);
  table.rows.forEach(function (r) { const id = tripText_(r[0]); counts[id] = (counts[id] || 0) + 1; });
  return table.rows.filter(function (r) { return allowed.includes(tripText_(r[0])) && counts[tripText_(r[0])] === 1; })
    .map(function (r) {
      const id = tripText_(r[0]), names = { 'R01-AC': '福森號第1梯次', 'R01-BD': '福森號第2梯次', R02: '台中南投' };
      return { id: id, name: names[id] };
    });
}

function tripDiningDiet_(value) {
  const key = tripText_(value).normalize('NFKC').replace(/\s+/g, '').toLowerCase();
  if (['素食', '素', '全素', '蛋奶素', '奶蛋素', '蛋素', '奶素', 'vegetarian', 'vegan', '1', '是', 'yes', 'true', '✓', '✔', 'v'].includes(key)) return '素食';
  if (['', '葷', '葷食', '一般', '無', '否', '0', 'no', 'false'].includes(key)) return '';
  return null;
}

function tripDiningData_(ss) {
  const sheet = ss.getSheetByName('用餐安排');
  if (!sheet) return { sheet: null, rows: [], snapshot: 'missing' };
  const range = sheet.getDataRange(), displayed = range.getDisplayValues(), formulas = range.getFormulas();
  if (TRIP_DINING_HEADERS.some(function (h, i) { return tripText_((displayed[0] || [])[i]) !== h; }) ||
      formulas.some(function (row) { return row.some(function (v) { return !!v; }); })) throw new Error('Invalid dining schema');
  const rows = displayed.slice(1).map(function (r) { return TRIP_DINING_HEADERS.map(function (_, i) { return tripText_(r[i]); }); });
  while (rows.length && !rows[rows.length - 1].some(Boolean)) rows.pop();
  return { sheet: sheet, rows: rows, snapshot: tripDigest_(JSON.stringify(rows)) };
}

function tripDiningEmployeeSnapshot_(rows) {
  return tripDigest_(JSON.stringify(rows.map(function (r) {
    return [tripParticipantKey_(r), tripText_(r[1]), tripText_(r[2]), tripText_(r[4])];
  })));
}

function tripDiningCandidatesFromTables_(tables) {
  const candidates = [], issues = [], routes = [];
  let headerFound = false;
  tables.forEach(function (rows, sheetIndex) {
    let header = null;
    for (let i = 0; i < Math.min(rows.length, 30); i++) {
      const cells = rows[i] || [], types = cells.map(tripRoomHeaderKey_);
      const name = types.indexOf('姓名'), table = types.indexOf('桌號') >= 0 ? types.indexOf('桌號') : types.indexOf('桌次');
      if (name >= 0 && table >= 0) {
        const diet = types.findIndex(function (key) { return ['飲食習慣', '飲食', '素食'].includes(key); });
        header = { row: i, name: name, table: table, seat: types.indexOf('座位序號'), diet: diet }; break;
      }
      if (['桌次人數', '桌次', '桌號'].includes(types[0]) && cells.slice(1).some(function (v) { return !!tripText_(v); })) {
        header = { row: i, horizontal: true }; break;
      }
    }
    if (!header) return;
    headerFound = true;
    for (let i = 0; i < header.row; i++) {
      const r = rows[i] || [], key = tripRoomHeaderKey_(r[0]);
      if (['行程', '路線', '路線id', '參加方案'].includes(key)) {
        const value = tripText_(r[1]);
        if (!value) continue;
        const route = tripRosterRoute_(value);
        if (!route || !TRIP_FUSEN_ROUTES.concat(['R02']).includes(route)) routes.push('invalid');
        else if (!routes.includes(route)) routes.push(route);
      }
    }
    const tableColumns = Object.create(null), duplicateTables = Object.create(null);
    if (header.horizontal) {
      (rows[header.row] || []).slice(1, TRIP_ROOM_IMPORT_MAX_COLUMNS).forEach(function (value, index) {
        const table = tripRoomValue_(value), column = index + 1;
        if (!table) {
          if (tripText_(value)) issues.push({ source: '工作表' + (sheetIndex + 1), value: tripText_(value), reason: 'invalid_table' });
          return;
        }
        if (Object.keys(tableColumns).some(function (c) { return tableColumns[c] === table; })) duplicateTables[table] = true;
        tableColumns[column] = table;
      });
    }
    for (let i = header.row + 1; i < rows.length; i++) {
      const row = rows[i] || [], source = '工作表' + (sheetIndex + 1) + ' 第' + (i + 1) + '列';
      if (header.horizontal && ['合計', '總計', '人數', '備註'].includes(tripRoomHeaderKey_(row[0]))) continue;
      const append = function (rawName, rawTable, rawSeat, duplicate, column) {
        if (!tripText_(rawName)) return;
        const name = tripRoomValue_(rawName), table = tripRoomValue_(rawTable), seat = tripRoomValue_(rawSeat);
        const colored = typeof tripDiningCellVegetarian_ === 'function' && tripDiningCellVegetarian_(rows, i, column);
        const explicitDiet = !header.horizontal && header.diet >= 0 ? tripDiningDiet_(row[header.diet]) : '';
        const diet = colored ? '素食' : explicitDiet;
        let reason = duplicate ? 'duplicate_table_header' : !name || /^\d+$/.test(name) ? 'invalid_name' : !table ? 'invalid_table' : '';
        if (!reason && tripText_(rawSeat) && (!seat || !/^\d{1,3}$/.test(seat))) reason = 'invalid_seat';
        if (!reason && diet === null) reason = 'invalid_diet';
        if (reason) issues.push({ source: source, value: tripText_(rawName), reason: reason });
        else candidates.push({ source: source, name: name, table: table, seat: seat, diet: diet });
        if (candidates.length + issues.length > TRIP_ROOM_IMPORT_MAX_ASSIGNMENTS) throw new Error('Too many dining rows');
      };
      if (header.horizontal) Object.keys(tableColumns).forEach(function (column) {
        append(row[Number(column)], tableColumns[column], row[0], duplicateTables[tableColumns[column]], Number(column));
      });
      else append(row[header.name], row[header.table], header.seat >= 0 ? row[header.seat] : '', false, header.name);
    }
  });
  return { headerFound: headerFound, candidates: candidates, issues: issues, routeId: routes.length === 1 ? routes[0] : '',
    error: routes.includes('invalid') || routes.length > 1 ? 'invalid_dining_route' : '' };
}

function tripDiningStart_(ss, ownerHash, parsed, forceRouteChoice) {
  if (!parsed.candidates.length) return { success: true, status: 'no_rows', kind: 'dining',
    summary: ['沒有可匯入的桌次資料，請確認桌號與姓名。', ...tripDiningIssueLines_(parsed.issues || [])].join('\n').slice(0, 1800),
    issueCount: (parsed.issues || []).length };
  let routes = tripDiningRoutes_(ss);
  if (parsed.routeId) {
    if (!routes.some(function (r) { return r.id === parsed.routeId; })) return tripStatus_('invalid_dining_route');
    if (!forceRouteChoice) return tripDiningPreview_(ss, ownerHash, parsed, parsed.routeId);
    routes = routes.filter(function (r) { return r.id === parsed.routeId; });
  }
  if (!routes.length) return tripStatus_('invalid_dining_route');
  const importId = tripRoomCacheSave_(ownerHash, [], 'dining', { stage: 'choose_route', parsed: parsed });
  if (!importId) return tripStatus_('preview_too_large');
  return { success: true, status: 'choose_route', kind: 'dining', importId: importId,
    expiresInSeconds: TRIP_ROOM_IMPORT_TTL_SECONDS, routes: routes,
    summary: '已辨識桌次表，請選擇這份桌次適用的行程。選擇後會先預覽，不會立即寫入。' };
}

function tripDiningSelectRoute_(ss, access, request) {
  if (!tripRoomImportId_(request.importId) || typeof request.routeId !== 'string') return tripStatus_('invalid_request');
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(4000)) return tripStatus_('busy');
  try {
    const currentAccess = tripRoomAdminAccess_(ss, request.lineUserId);
    if (!currentAccess || currentAccess.ownerHash !== access.ownerHash) return tripStatus_('forbidden');
    const saved = tripRoomCacheRead_(request.importId, access.ownerHash);
    if (!saved) return tripStatus_('expired');
    if (saved.kind !== 'dining' || !saved.metadata || saved.metadata.stage !== 'choose_route') return tripStatus_('invalid_request');
    const queue = tripImportQueueRead_(ss, saved, access.ownerHash);
    if (!queue) return tripStatus_('expired');
    if (saved.metadata.parsed.routeId && saved.metadata.parsed.routeId !== request.routeId) return tripStatus_('invalid_dining_route');
    if (!tripDiningRoutes_(ss).some(function (r) { return r.id === request.routeId; })) return tripStatus_('invalid_dining_route');
    const result = tripDiningPreview_(ss, access.ownerHash, saved.metadata.parsed, request.routeId);
    const history = saved.metadata.importHistory || [];
    if (result.status === 'preview') {
      if (!tripImportAttachQueue_(access.ownerHash, result, queue, history)) {
        CacheService.getScriptCache().remove(tripRoomImportKey_(result.importId));
        return tripStatus_('preview_too_large');
      }
      CacheService.getScriptCache().remove(tripRoomImportKey_(request.importId));
      return tripImportHistoryResult_(result, history);
    }
    if (result.status === 'no_rows') {
      CacheService.getScriptCache().remove(tripRoomImportKey_(request.importId));
      return tripImportAdvance_(ss, access.ownerHash, queue, result, history);
    }
    return result;
  } finally { lock.releaseLock(); }
}

function tripDiningIssueLines_(issues) {
  const labels = { invalid_name: '姓名格式錯誤', invalid_table: '桌號格式錯誤', invalid_seat: '座位序號格式錯誤', invalid_diet: '飲食習慣無法判定',
    duplicate_table_header: '桌號欄位重複', employee_not_found: '查無此人', ambiguous_employee: '同名人員無法判定',
    employee_disabled: '人員已停用', route_mismatch: '不屬於這個行程', conflicting_dining: '同一人重複安排座位',
    duplicate_seat: '同一座位有多位人員' };
  return issues.slice(0, 6).map(function (issue) { return tripMessageText_(issue.value || issue.source, 60) + '：' + (labels[issue.reason] || '未匯入'); });
}

function tripDiningPreview_(ss, ownerHash, parsed, routeId) {
  const data = tripDiningData_(ss), employees = tripEmployees_(ss).rows;
  const names = Object.create(null), keys = Object.create(null), selected = Object.create(null), occupied = Object.create(null);
  employees.forEach(function (row) {
    const name = tripRoomNormalizeName_(row[1]), key = tripParticipantKey_(row);
    if (!names[name]) names[name] = [];
    names[name].push(row);
    if (key) keys[key] = (keys[key] || 0) + 1;
  });
  const issues = (parsed.issues || []).slice();
  parsed.candidates.forEach(function (candidate) {
    const all = names[tripRoomNormalizeName_(candidate.name)] || [];
    const matches = all.filter(function (r) { const ids = tripEmployeeRouteIds_(r[2]); return ids && ids.includes(routeId); });
    const row = matches[0], key = row && tripParticipantKey_(row);
    const reason = !all.length ? 'employee_not_found' : !matches.length ? 'route_mismatch' : matches.length !== 1 ? 'ambiguous_employee' :
      tripText_(row[4]) !== '啟用' ? 'employee_disabled' : !key || keys[key] !== 1 ? 'ambiguous_employee' : '';
    if (reason) { issues.push({ source: candidate.source, value: candidate.name, reason: reason }); return; }
    const entry = { participantKey: key, name: tripText_(row[1]), table: candidate.table, seat: candidate.seat,
      diet: candidate.diet || '', conflict: false };
    if (selected[key]) {
      selected[key].conflict = true;
      entry.conflict = true;
      issues.push({ source: candidate.source, value: candidate.name, reason: 'conflicting_dining' });
    } else selected[key] = entry;
    if (candidate.seat) {
      const seatKey = candidate.table + '\u0000' + candidate.seat;
      if (occupied[seatKey] && occupied[seatKey] !== key) {
        selected[occupied[seatKey]].conflict = true;
        selected[key].conflict = true;
        issues.push({ source: candidate.source, value: candidate.name, reason: 'duplicate_seat' });
      } else occupied[seatKey] = key;
    }
  });
  const assignments = Object.keys(selected).filter(function (key) { return !selected[key].conflict; }).map(function (key) {
    const r = selected[key]; return { participantKey: r.participantKey, name: r.name, table: r.table, seat: r.seat, diet: r.diet };
  });
  const previous = data.rows.filter(function (r) { return r[1] === routeId; });
  const retained = data.rows.filter(function (r) { return r[1] !== routeId && r.some(Boolean); }).length;
  const removed = previous.filter(function (r) { return !assignments.some(function (a) { return a.participantKey === r[0]; }); }).length;
  const route = tripDiningRoutes_(ss).filter(function (r) { return r.id === routeId; })[0];
  if (!route) return tripStatus_('invalid_dining_route');
  const tableCounts = Object.create(null);
  assignments.forEach(function (a) { tableCounts[a.table] = (tableCounts[a.table] || 0) + 1; });
  const lines = ['桌次匯入預覽｜' + tripMessageText_(route.name, 150),
    '確認後將取代這個行程的整份原桌次；其他行程保留。',
    '原安排 ' + previous.length + ' 人 → 新安排 ' + assignments.length + ' 人',
    '未匹配／未匯入 ' + issues.length + ' 筆；移除原安排 ' + removed + ' 人；其他行程保留 ' + retained + ' 筆',
    '素食 ' + assignments.filter(function (a) { return a.diet === '素食'; }).length + ' 人',
    ...Object.keys(tableCounts).slice(0, 30).map(function (table) { return table + '：' + tableCounts[table] + ' 人'; })];
  lines.push('', ...assignments.slice(0, 8).map(function (a) { return a.name + (a.diet === '素食' ? '(素食)' : '') + ' → ' + a.table; }));
  if (issues.length) lines.push('', ...tripDiningIssueLines_(issues));
  if (!assignments.length) return { success: true, status: 'no_rows', kind: 'dining',
    summary: ['沒有可寫入的桌次；原安排保持不變。', ...lines].join('\n').slice(0, 1800),
    matchedCount: 0, issueCount: issues.length, issues: issues.slice(0, 100) };
  const importId = tripRoomCacheSave_(ownerHash, assignments, 'dining', { stage: 'preview', routeId: routeId,
    diningSnapshot: data.snapshot, employeeSnapshot: tripDiningEmployeeSnapshot_(employees) });
  if (!importId) return tripStatus_('preview_too_large');
  return { success: true, status: 'preview', kind: 'dining', importId: importId, canConfirm: true,
    expiresInSeconds: TRIP_ROOM_IMPORT_TTL_SECONDS, summary: lines.join('\n').slice(0, 1800),
    matchedCount: assignments.length, issueCount: issues.length, removedCount: removed,
    previousCount: previous.length, issues: issues.slice(0, 100) };
}

function tripDiningConfirmSaved_(ss, saved, importId) {
  const meta = saved.metadata;
  if (!meta || meta.stage !== 'preview' || !TRIP_FUSEN_ROUTES.concat(['R02']).includes(meta.routeId) ||
      !saved.assignments.length || saved.assignments.length > TRIP_ROOM_IMPORT_MAX_ASSIGNMENTS) return tripStatus_('invalid_request');
  if (!tripDiningRoutes_(ss).some(function (r) { return r.id === meta.routeId; })) return tripStatus_('stale');
  const data = tripDiningData_(ss), employees = tripEmployees_(ss).rows;
  if (data.snapshot !== meta.diningSnapshot || tripDiningEmployeeSnapshot_(employees) !== meta.employeeSnapshot) return tripStatus_('stale');
  const seen = Object.create(null), seats = Object.create(null), replacement = [];
  for (let i = 0; i < saved.assignments.length; i++) {
    const a = saved.assignments[i], matches = employees.filter(function (r) { return tripParticipantKey_(r) === a.participantKey; });
    const row = matches[0], ids = row && tripEmployeeRouteIds_(row[2]);
    if (matches.length !== 1 || tripText_(row[1]) !== a.name || tripText_(row[4]) !== '啟用' || !ids || !ids.includes(meta.routeId) ||
        seen[a.participantKey] || !tripRoomValue_(a.table) || tripRoomValue_(a.table) !== a.table ||
        (a.seat && !/^\d{1,3}$/.test(a.seat)) || !['', '素食'].includes(a.diet || '')) return tripStatus_('stale');
    const seatKey = a.table + '\u0000' + a.seat;
    if (a.seat && seats[seatKey]) return tripStatus_('invalid_request');
    seats[seatKey] = true; seen[a.participantKey] = true;
    replacement.push([a.participantKey, meta.routeId, a.table, a.seat, a.name, a.diet || '']);
  }
  const kept = data.rows.filter(function (r) { return r[1] !== meta.routeId && r.some(Boolean); });
  const rows = [TRIP_DINING_HEADERS].concat(kept, replacement), oldHeight = data.rows.length + 1;
  while (rows.length < oldHeight) rows.push(['', '', '', '', '', '']);
  const sheet = data.sheet || ss.insertSheet('用餐安排');
  try {
    if (rows.length > sheet.getMaxRows()) sheet.insertRowsAfter(sheet.getMaxRows(), rows.length - sheet.getMaxRows());
    sheet.getRange(1, 1, rows.length, TRIP_DINING_HEADERS.length).setValues(rows);
  } catch (error) {
    // The route plan is written in one call. If its first write fails, do not
    // leave an empty, invalid new table blocking all later seating queries.
    if (!data.sheet) ss.deleteSheet(sheet);
    throw error;
  }
  SpreadsheetApp.flush();
  CacheService.getScriptCache().remove(tripRoomImportKey_(importId));
  return { success: true, status: 'ok', kind: 'dining', updatedCount: replacement.length,
    summary: '已更新 ' + replacement.length + ' 人的桌次。其他行程的桌次保留。' };
}

function tripDiningReply_(ss, access, routeIds) {
  const data = tripDiningData_(ss), employees = tripEmployees_(ss).rows;
  const ownRows = employees.filter(function (r) {
    return access.employee.id ? tripEmployeeId_(r[0]) === access.employee.id :
      !tripEmployeeId_(r[0]) && tripRoomNormalizeName_(r[1]) === tripRoomNormalizeName_(access.employee.name);
  });
  if (ownRows.length !== 1) return '';
  const own = ownRows[0], ownKey = tripParticipantKey_(own), keys = Object.create(null), routeNames = Object.create(null);
  if (!ownKey) return '';
  employees.forEach(function (r) { const key = tripParticipantKey_(r); if (key) { if (!keys[key]) keys[key] = []; keys[key].push(r); } });
  tripDiningRoutes_(ss).forEach(function (r) { routeNames[r.id] = r.name; });
  const sections = [];
  routeIds.forEach(function (routeId) {
    if (!TRIP_FUSEN_ROUTES.concat(['R02']).includes(routeId)) return;
    let seat = '', diet = '', colleagues = [];
    const rows = data.rows.filter(function (r) { return r[1] === routeId; }), hasRoutePlan = rows.length > 0;
    if (hasRoutePlan) {
      const ownDining = rows.filter(function (r) { return r[0] === ownKey; });
      if (ownDining.length === 1 && tripRoomNormalizeName_(ownDining[0][4]) === tripRoomNormalizeName_(own[1])) {
        seat = tripRoomValue_(ownDining[0][2]);
        diet = ownDining[0][5] === '素食' ? '素食' : '';
        if (seat) rows.forEach(function (r) {
          if (r[0] === ownKey || r[2] !== seat || rows.filter(function (q) { return q[0] === r[0]; }).length !== 1) return;
          const matches = keys[r[0]] || [], person = matches[0], ids = person && tripEmployeeRouteIds_(person[2]);
          if (matches.length === 1 && tripText_(person[4]) === '啟用' && ids && ids.includes(routeId) &&
              tripRoomNormalizeName_(person[1]) === tripRoomNormalizeName_(r[4])) colleagues.push(tripText_(person[1]));
        });
      }
    } else {
      // H predates multiple routes and cannot identify which journey a seat
      // belongs to. Use it only when both people have one explicit route.
      const ownRouteIds = tripEmployeeRouteIds_(own[2]);
      seat = ownRouteIds && ownRouteIds.length === 1 ? tripRoomValue_(own[7]) : '';
      if (seat) employees.forEach(function (r) {
        const key = tripParticipantKey_(r), ids = tripEmployeeRouteIds_(r[2]);
        if (key && key !== ownKey && (keys[key] || []).length === 1 && tripText_(r[4]) === '啟用' && ids && ids.length === 1 && ids.includes(routeId) &&
            tripRoomValue_(r[7]) === seat) colleagues.push(tripText_(r[1]));
      });
    }
    // An absent legacy seat keeps the existing image-only experience.
    if (!hasRoutePlan && !seat) return;
    let text = seat ? '您今日的用餐座位為：' + seat : '您的用餐座位尚未安排。';
    if (seat && diet) text += '\n飲食：素食';
    if (seat && colleagues.length) text += '\n您與' + colleagues.slice(0, 50).map(function (n) { return tripMessageText_(n, 50); }).join('、') +
      (colleagues.length > 50 ? '等人' : '') + '同桌。';
    if (routeIds.length > 1) text = '【' + tripMessageText_(routeNames[routeId] || routeId, 150) + '】\n' + text;
    sections.push(text);
  });
  return sections.join('\n\n').slice(0, 3800);
}

// Vehicle plans are separate per journey, so A+B participants keep both cars.
const TRIP_VEHICLE_HEADERS = ['人員識別碼', '路線ID', '車次', '姓名'];

function tripVehicleRoute_(value, batch) {
  const label = tripText_(value).normalize('NFKC').replace(/\s+/g, '')
    .replace(/\(\d+人\)$/, '');
  let route = tripRosterRoute_(label);
  const combo = tripRosterFusenCombo_(batch);
  if (combo === null || !route || route.indexOf(',') >= 0) return '';
  if (combo) {
    if (route === 'R01') route = 'R01-' + combo;
    else if (route !== 'R01-' + combo) return '';
  }
  return TRIP_FUSEN_ROUTES.concat(['R02']).includes(route) ? route : '';
}

function tripVehicleCandidatesFromTables_(tables) {
  const candidates = [], issues = [], routeIds = [], vehicleTableIndices = [];
  let headerFound = false;
  tables.forEach(function (rows, sheetIndex) {
    let header = null;
    for (let rowIndex = 0; rowIndex < Math.min(rows.length, 30); rowIndex++) {
      const cells = rows[rowIndex] || [], types = cells.map(tripRoomHeaderKey_);
      const names = types.map(function (value, column) { return tripRoomHeaderType_(cells[column]) === 'name' ? column : -1; }).filter(function (i) { return i >= 0; });
      const vehicle = types.findIndex(function (key) { return ['車次', '車號', '乘車車次', '班次', 'vehicle', 'coach'].includes(key); });
      if (names.length === 1 && vehicle >= 0) {
        const route = types.findIndex(function (key, column) { return key === '行程' || tripRosterHeaderType_(cells[column]) === 'route'; });
        const batch = types.findIndex(function (_, column) { return tripRosterHeaderType_(cells[column]) === 'fusenCombo'; });
        header = { row: rowIndex, name: names[0], vehicle: vehicle, route: route, batch: batch }; break;
      }
      if (['人數', '乘車人數', '序號', '編號'].includes(types[0]) && cells.slice(1).some(function (value) {
        return /車$/.test(tripText_(value)) && !['車次', '車號', '車輛'].includes(tripText_(value));
      })) { header = { row: rowIndex, horizontal: true }; break; }
    }
    if (!header) return;
    headerFound = true; vehicleTableIndices.push(sheetIndex);
    let defaultRoute = '';
    for (let i = 0; i < header.row; i++) {
      const row = rows[i] || [], key = tripRoomHeaderKey_(row[0]);
      if (['行程', '路線', '路線id', '參加方案'].includes(key) && tripText_(row[1])) defaultRoute = tripVehicleRoute_(row[1]);
    }
    const columns = Object.create(null), seenVehicles = Object.create(null);
    if (header.horizontal) {
      (rows[header.row] || []).slice(1, TRIP_ROOM_IMPORT_MAX_COLUMNS).forEach(function (rawVehicle, offset) {
        if (!tripText_(rawVehicle)) return;
        const column = offset + 1, vehicle = tripRoomValue_(rawVehicle);
        let routeId = '';
        for (let i = header.row - 1; i >= 0 && !routeId; i--) routeId = tripVehicleRoute_((rows[i] || [])[column]);
        if (!routeId) routeId = defaultRoute;
        const reason = !vehicle ? 'invalid_vehicle' : !routeId ? 'invalid_vehicle_route' : '';
        if (reason) {
          issues.push({ source: '工作表' + (sheetIndex + 1) + ' 第' + (column + 1) + '欄', value: tripText_(rawVehicle), reason: reason });
          return;
        }
        const key = routeId + '\u0000' + vehicle;
        if (seenVehicles[key]) seenVehicles[key].duplicate = true;
        const entry = { routeId: routeId, vehicle: vehicle, duplicate: !!seenVehicles[key] };
        columns[column] = entry;
        if (!seenVehicles[key]) seenVehicles[key] = entry;
        if (!routeIds.includes(routeId)) routeIds.push(routeId);
      });
    }
    for (let i = header.row + 1; i < rows.length; i++) {
      const row = rows[i] || [], source = '工作表' + (sheetIndex + 1) + ' 第' + (i + 1) + '列';
      if (header.horizontal && ['合計', '總計', '人數', '備註'].includes(tripRoomHeaderKey_(row[0]))) continue;
      function append(rawName, routeId, rawVehicle, duplicate) {
        if (!tripText_(rawName)) return;
        const name = tripRoomValue_(rawName), vehicle = tripRoomValue_(rawVehicle);
        const reason = !name || /^\d+$/.test(name) ? 'invalid_name' : !routeId ? 'invalid_vehicle_route' :
          !vehicle ? 'invalid_vehicle' : duplicate ? 'duplicate_vehicle_header' : '';
        if (reason) issues.push({ source: source, value: tripText_(rawName), reason: reason });
        else {
          candidates.push({ name: name, routeId: routeId, vehicle: vehicle, source: source });
          if (!routeIds.includes(routeId)) routeIds.push(routeId);
        }
        if (candidates.length + issues.length > TRIP_ROOM_IMPORT_MAX_ASSIGNMENTS) throw new Error('Too many vehicle rows');
      }
      if (header.horizontal) Object.keys(columns).forEach(function (column) {
        const entry = columns[column]; append(row[Number(column)], entry.routeId, entry.vehicle, entry.duplicate);
      });
      else {
        if (!tripText_(row[header.name]) && !tripText_(row[header.vehicle])) continue;
        const routeId = header.route >= 0 ? tripVehicleRoute_(row[header.route], header.batch >= 0 ? row[header.batch] : '') : defaultRoute;
        append(row[header.name], routeId, row[header.vehicle], false);
      }
    }
  });
  return { headerFound: headerFound, candidates: candidates, issues: issues, routeIds: routeIds, vehicleTableIndices: vehicleTableIndices };
}

function tripVehicleData_(ss) {
  const sheet = ss.getSheetByName('乘車安排');
  if (!sheet) return { sheet: null, rows: [], snapshot: 'missing' };
  const range = sheet.getDataRange(), displayed = range.getDisplayValues(), formulas = range.getFormulas();
  if (TRIP_VEHICLE_HEADERS.some(function (header, i) { return tripText_((displayed[0] || [])[i]) !== header; }) ||
      formulas.some(function (row) { return row.some(function (value) { return !!value; }); })) throw new Error('Invalid vehicle schema');
  const rows = displayed.slice(1).map(function (row) { return TRIP_VEHICLE_HEADERS.map(function (_, i) { return tripText_(row[i]); }); });
  while (rows.length && !rows[rows.length - 1].some(Boolean)) rows.pop();
  return { sheet: sheet, rows: rows, snapshot: tripDigest_(JSON.stringify(rows)) };
}

function tripVehicleIssueLines_(issues) {
  const labels = { invalid_name: '姓名格式錯誤', invalid_vehicle: '車次格式錯誤', invalid_vehicle_route: '請明確填寫行程及福森號梯次',
    duplicate_vehicle_header: '同一行程的車號欄位重複', employee_not_found: '查無此人', ambiguous_employee: '同名或人員識別資料無法判定',
    employee_disabled: '人員已停用', route_mismatch: '不屬於這個行程', conflicting_vehicle: '同一人同一行程重複安排車次' };
  return issues.slice(0, 6).map(function (issue) {
    return tripMessageText_(issue.value || issue.source, 60) + '：' + (labels[issue.reason] || '未匯入');
  });
}

function tripVehicleStart_(ss, ownerHash, parsed) {
  const data = tripVehicleData_(ss), employees = tripEmployees_(ss).rows;
  const allowed = tripDiningRoutes_(ss).map(function (route) { return route.id; });
  const names = Object.create(null), keys = Object.create(null), selected = Object.create(null);
  employees.forEach(function (row) {
    const name = tripRoomNormalizeName_(row[1]), key = tripParticipantKey_(row);
    if (!names[name]) names[name] = [];
    names[name].push(row);
    if (key) keys[key] = (keys[key] || 0) + 1;
  });
  const issues = (parsed.issues || []).slice();
  (parsed.candidates || []).forEach(function (candidate) {
    const matches = names[tripRoomNormalizeName_(candidate.name)] || [], row = matches[0];
    const key = row && tripParticipantKey_(row), routes = row && tripEmployeeRouteIds_(row[2]);
    const reason = !allowed.includes(candidate.routeId) ? 'invalid_vehicle_route' : !matches.length ? 'employee_not_found' :
      matches.length !== 1 || !key || keys[key] !== 1 ? 'ambiguous_employee' : tripText_(row[4]) !== '啟用' ? 'employee_disabled' :
      !routes || !routes.includes(candidate.routeId) ? 'route_mismatch' : !tripRoomValue_(candidate.vehicle) ? 'invalid_vehicle' : '';
    if (reason) { issues.push({ source: candidate.source, value: candidate.name, reason: reason }); return; }
    const assignmentKey = candidate.routeId + '\u0000' + key;
    if (selected[assignmentKey]) {
      selected[assignmentKey].conflict = true;
      issues.push({ source: candidate.source, value: candidate.name, reason: 'conflicting_vehicle' });
      return;
    }
    selected[assignmentKey] = { participantKey: key, name: tripText_(row[1]), routeId: candidate.routeId, vehicle: tripRoomValue_(candidate.vehicle) };
  });
  const assignments = Object.keys(selected).filter(function (key) { return !selected[key].conflict; }).map(function (key) { return selected[key]; });
  const routeIds = allowed.filter(function (routeId) { return assignments.some(function (a) { return a.routeId === routeId; }); });
  const previous = data.rows.filter(function (row) { return routeIds.includes(row[1]); });
  const retained = data.rows.filter(function (row) { return !routeIds.includes(row[1]) && row.some(Boolean); }).length;
  const removed = previous.filter(function (row) { return !assignments.some(function (a) { return a.routeId === row[1] && a.participantKey === row[0]; }); }).length;
  const vehicleCounts = Object.create(null);
  assignments.forEach(function (a) { const key = tripRouteLabel_(a.routeId) + '｜' + a.vehicle; vehicleCounts[key] = (vehicleCounts[key] || 0) + 1; });
  const skippedRoutes = (parsed.routeIds || []).filter(function (id) { return !routeIds.includes(id); });
  const lines = ['車次匯入預覽', '確認後將取代下列行程的整份原車次；其他行程保留。',
    routeIds.map(tripRouteLabel_).join('、') || '沒有可更新的行程',
    '原安排 ' + previous.length + ' 筆 → 新安排 ' + assignments.length + ' 筆',
    '未匹配／未匯入 ' + issues.length + ' 筆；移除原安排 ' + removed + ' 筆；其他行程保留 ' + retained + ' 筆',
    ...Object.keys(vehicleCounts).slice(0, 30).map(function (key) { return key + '：' + vehicleCounts[key] + ' 人'; })];
  if (skippedRoutes.length) lines.push('以下行程沒有可配對名單，原車次保留：' + skippedRoutes.map(tripRouteLabel_).join('、'));
  lines.push('', ...assignments.slice(0, 8).map(function (a) { return a.name + ' → ' + tripRouteLabel_(a.routeId) + ' ' + a.vehicle; }));
  if (issues.length) lines.push('', ...tripVehicleIssueLines_(issues));
  if (!assignments.length) return { success: true, status: 'no_rows', kind: 'vehicle', summary: ['沒有可寫入的車次；原安排保持不變。', ...lines].join('\n').slice(0, 1800),
    matchedCount: 0, issueCount: issues.length, issues: issues.slice(0, 100) };
  const importId = tripRoomCacheSave_(ownerHash, assignments, 'vehicle', { stage: 'preview', routeIds: routeIds,
    vehicleSnapshot: data.snapshot, employeeSnapshot: tripDiningEmployeeSnapshot_(employees) });
  if (!importId) return tripStatus_('preview_too_large');
  return { success: true, status: 'preview', kind: 'vehicle', importId: importId, canConfirm: true,
    expiresInSeconds: TRIP_ROOM_IMPORT_TTL_SECONDS, summary: lines.join('\n').slice(0, 1800),
    matchedCount: assignments.length, issueCount: issues.length, routeIds: routeIds, previousCount: previous.length,
    removedCount: removed, issues: issues.slice(0, 100) };
}

function tripVehicleConfirmSaved_(ss, saved, importId) {
  const meta = saved.metadata, allowed = tripDiningRoutes_(ss).map(function (route) { return route.id; });
  if (!meta || meta.stage !== 'preview' || !Array.isArray(meta.routeIds) || !meta.routeIds.length ||
      meta.routeIds.length > 3 || new Set(meta.routeIds).size !== meta.routeIds.length ||
      meta.routeIds.some(function (id) { return !TRIP_FUSEN_ROUTES.concat(['R02']).includes(id); }) || !saved.assignments.length ||
      saved.assignments.length > TRIP_ROOM_IMPORT_MAX_ASSIGNMENTS) return tripStatus_('invalid_request');
  if (meta.routeIds.some(function (id) { return !allowed.includes(id); })) return tripStatus_('stale');
  const data = tripVehicleData_(ss), employees = tripEmployees_(ss).rows;
  if (data.snapshot !== meta.vehicleSnapshot || tripDiningEmployeeSnapshot_(employees) !== meta.employeeSnapshot) return tripStatus_('stale');
  const seen = Object.create(null), replacement = [];
  for (let i = 0; i < saved.assignments.length; i++) {
    const a = saved.assignments[i], matches = employees.filter(function (row) { return tripParticipantKey_(row) === a.participantKey; });
    const row = matches[0], routes = row && tripEmployeeRouteIds_(row[2]), key = a.routeId + '\u0000' + a.participantKey;
    if (!meta.routeIds.includes(a.routeId) || matches.length !== 1 || tripText_(row[1]) !== a.name || tripText_(row[4]) !== '啟用' ||
        !routes || !routes.includes(a.routeId) || seen[key] || !tripRoomValue_(a.vehicle) || tripRoomValue_(a.vehicle) !== a.vehicle) return tripStatus_('stale');
    seen[key] = true; replacement.push([a.participantKey, a.routeId, a.vehicle, a.name]);
  }
  if (meta.routeIds.some(function (routeId) { return !replacement.some(function (row) { return row[1] === routeId; }); })) return tripStatus_('invalid_request');
  const kept = data.rows.filter(function (row) { return !meta.routeIds.includes(row[1]) && row.some(Boolean); });
  const rows = [TRIP_VEHICLE_HEADERS].concat(kept, replacement), oldHeight = data.rows.length + 1;
  while (rows.length < oldHeight) rows.push(['', '', '', '']);
  const sheet = data.sheet || ss.insertSheet('乘車安排');
  try {
    if (rows.length > sheet.getMaxRows()) sheet.insertRowsAfter(sheet.getMaxRows(), rows.length - sheet.getMaxRows());
    sheet.getRange(1, 1, rows.length, TRIP_VEHICLE_HEADERS.length).setValues(rows);
  } catch (error) { if (!data.sheet) ss.deleteSheet(sheet); throw error; }
  SpreadsheetApp.flush();
  CacheService.getScriptCache().remove(tripRoomImportKey_(importId));
  return { success: true, status: 'ok', kind: 'vehicle', updatedCount: replacement.length,
    summary: '已更新 ' + replacement.length + ' 筆乘車安排。其他行程的車次保留。' };
}

function tripVehicleReply_(ss, access, routeIds) {
  if (!access || access.status !== 'approved' || !access.employee || !Array.isArray(routeIds)) return '您的車次尚未安排。';
  const data = tripVehicleData_(ss), employees = tripEmployees_(ss).rows;
  const ownRows = employees.filter(function (row) {
    return access.employee.id ? tripEmployeeId_(row[0]) === access.employee.id :
      !tripEmployeeId_(row[0]) && tripRoomNormalizeName_(row[1]) === tripRoomNormalizeName_(access.employee.name);
  });
  if (ownRows.length !== 1 || tripText_(ownRows[0][4]) !== '啟用') return '您的車次尚未安排。';
  const own = ownRows[0], key = tripParticipantKey_(own), ownRoutes = tripEmployeeRouteIds_(own[2]);
  if (!key || !ownRoutes || employees.filter(function (row) { return tripParticipantKey_(row) === key; }).length !== 1) return '您的車次尚未安排。';
  const sections = [];
  (routeIds || []).forEach(function (routeId) {
    if (!ownRoutes.includes(routeId)) return;
    if (routeId === 'R01') { sections.push(TRIP_FUSEN_PENDING); return; }
    if (!TRIP_FUSEN_ROUTES.concat(['R02']).includes(routeId)) return;
    const routeRows = data.rows.filter(function (row) { return row[1] === routeId; });
    const assignments = routeRows.filter(function (row) { return row[0] === key; });
    let vehicle = '';
    if (routeRows.length) {
      if (assignments.length === 1 && tripRoomNormalizeName_(assignments[0][3]) === tripRoomNormalizeName_(own[1])) vehicle = tripRoomValue_(assignments[0][2]);
    } else if (ownRoutes.length === 1) vehicle = tripRoomValue_(own[6]);
    let text = vehicle ? '您的車次為：' + vehicle : '您的車次尚未安排。';
    if (routeIds.length > 1) text = '【' + tripRouteLabel_(routeId) + '】\n' + text;
    sections.push(text);
  });
  return sections.join('\n\n').slice(0, 3800) || '您的車次尚未安排。';
}
