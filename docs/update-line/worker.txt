  async fetch(request, env, ctx) {

      });

      return new Response('Unauthorized', { status: 401 });
    }
    let data;
    try { data = JSON.parse(body); } catch { return new Response('Bad Request', { status: 400 }); }
    if (!Array.isArray(data.events)) return new Response('Bad Request', { status: 400 });
    if (data.events.length === 0) return new Response('OK');
    if (!env.LINE_CHANNEL_ACCESS_TOKEN) {
      console.error('Missing LINE_CHANNEL_ACCESS_TOKEN');
      return new Response('Service Unavailable', { status: 503 });
    }
    // Start promptly, acknowledge the webhook, and keep work alive for the reply.
    const work = Promise.allSettled(data.events.map(event => handleEvent(event, env)));
    if (ctx?.waitUntil) ctx.waitUntil(work);

    return new Response('OK');
  },

async function validSignature(body, signature, secret) {
  if (!signature || !secret) return false;
  const encoder = new TextEncoder();
  const key = await crypto.subtle.importKey('raw', encoder.encode(secret),
    { name: 'HMAC', hash: 'SHA-256' }, false, ['sign']);
  const digest = await crypto.subtle.sign('HMAC', key, encoder.encode(body));
  const expected = btoa(String.fromCharCode(...new Uint8Array(digest)));
  if (signature.length !== expected.length) return false;
  let difference = 0;
  for (let i = 0; i < expected.length; i++) difference |= signature.charCodeAt(i) ^ expected.charCodeAt(i);
  return difference === 0;
}

async function handleEvent(event, env) {
  if (!['follow', 'message', 'postback'].includes(event.type) ||
      (event.type === 'message' && !['text', 'file'].includes(event.message?.type)) ||
      !event.replyToken || event.mode === 'standby') return;
  const now = Date.now();
  for (const [key, expires] of recentEvents) if (expires <= now) recentEvents.delete(key);
  const eventKey = event.webhookEventId || event.replyToken;
  if (recentEvents.has(eventKey)) return;
  recentEvents.set(eventKey, now + 20 * 60 * 1000);
  if (recentEvents.size > 5000) recentEvents.delete(recentEvents.keys().next().value);
  // New-friend greeting is managed in LINE Official Account Manager.
  // Only synchronize the appropriate menu here; do not send a second greeting.
  if (event.type === 'follow') {
    if (event.source?.type === 'user' && event.source.userId && env.APPS_SCRIPT_URL && env.API_SECRET) {
      try { await syncGuideMenu(event.source.userId, env); }
      catch (error) { logApiError('Guide menu sync failed', error); }
    }
    return;
  }
  if (event.type === 'message' && event.message?.type === 'file') {
    await handleRoomFileEvent(event, env, now);
    return;
  }
  let text;
  try {
    text = event.source?.type !== 'user' || !event.source.userId
      ? '為保護個人資料，請在官方帳號的一對一聊天室查詢行程。'
      : await buildReply(event.type === 'postback' ? (event.postback?.data || '') : event.message.text.trim(), event.source.userId, env, now + 18000);
  } catch (error) {
    // Never log request bodies, API responses, names, user IDs, tokens or URLs.
    const code = logApiError('Employee API request failed', error);
    text = `${RETRY}\n參考代碼：${code}`;
  }
  let messages;
  try {
    messages = lineReplyMessages(text);
  } catch (error) {
    const code = logApiError('LINE reply message validation failed', error);
    messages = [{ type: 'text', text: `${RETRY}\n參考代碼：${code}` }];
  }
  try {
    const response = await fetch('https://api.line.me/v2/bot/message/reply', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${env.LINE_CHANNEL_ACCESS_TOKEN}` },
      body: JSON.stringify({ replyToken: event.replyToken, messages }),
      signal: AbortSignal.timeout(8000)
    });
    if (!response.ok) console.error('LINE reply failed', response.status);
    else if (event.source?.type === 'user' && event.source.userId && env.APPS_SCRIPT_URL && env.API_SECRET) {
      try { await syncGuideMenu(event.source.userId, env); }
      catch (error) { logApiError('Guide menu sync failed', error); }
    }
  } catch { console.error('LINE reply request failed'); }
}

function lineReplyMessages(value) {
  const values = Array.isArray(value) ? value : [value];
  if (values.length < 1 || values.length > 5) throw new EmployeeApiError('LINE_MESSAGES_INVALID');
  return values.map(item => {
    if (typeof item === 'string') {
      if (!validText(item)) throw new EmployeeApiError('LINE_MESSAGES_INVALID');
      return { type: 'text', text: item };
    }
    if (!item || typeof item !== 'object' || Array.isArray(item)) throw new EmployeeApiError('LINE_MESSAGES_INVALID');
    if (item.type === 'text' && validText(item.text)) return item;
    if (item.type === 'image' && validImageUrl(item.originalContentUrl) && validImageUrl(item.previewImageUrl)) return item;
    throw new EmployeeApiError('LINE_MESSAGES_INVALID');
    return item;
  });
}

function rosterImportSummaryText(result, operation) {
  if (validText(result.summary)) return result.summary;
  const raw = result.counts && typeof result.counts === 'object' && !Array.isArray(result.counts)
    ? result.counts
    : result.summary && typeof result.summary === 'object' && !Array.isArray(result.summary)
      ? result.summary
      : result;
  const readCount = (...keys) => {
    for (const source of raw === result ? [raw] : [raw, result]) {
      for (const key of keys) {
        const value = source[key];
        if (Number.isSafeInteger(value) && value >= 0 && value <= 100000) return value;
      }
    }
    return null;
  };
  const counts = {
    input: readCount('input', 'total', 'inputCount', 'totalCount'),
    added: readCount('new', 'added', 'newCount', 'addedCount'),
    updated: readCount('updated', 'update', 'updatedCount', 'updateCount'),
    unchanged: readCount('unchanged', 'same', 'unchangedCount', 'sameCount'),
    conflict: readCount('conflict', 'conflicts', 'conflictCount', 'issueCount')
  };
  if (operation === 'cancel') return '已取消員工名單匯入。';
  if (Object.values(counts).every(value => value === null)) return '';
  const title = operation === 'confirm' ? '員工名單匯入完成' : '員工名單匯入預覽';
  const lines = [title];
  if (counts.input !== null) lines.push(`讀取 ${counts.input} 人`);
  if (counts.added !== null) lines.push(`新增 ${counts.added} 人`);
  if (counts.updated !== null) lines.push(`更新 ${counts.updated} 人`);
  if (counts.unchanged !== null) lines.push(`資料相同 ${counts.unchanged} 人`);
  if (counts.conflict !== null) lines.push(`需確認 ${counts.conflict} 人`);
  const samples = Array.isArray(result.samples) ? result.samples : Array.isArray(raw.samples) ? raw.samples : [];
  const safeSamples = samples.slice(0, 5).map(value => typeof value === 'string'
    ? value.replace(/[\u0000-\u001f\u007f]/g, ' ').trim().slice(0, 120)
    : '').filter(Boolean);
  if (safeSamples.length) lines.push('', '需確認的資料：', ...safeSamples.map(value => `- ${value}`));
  if (operation === 'previewFile') lines.push('', '請確認後寫入。');
  return lines.join('\n');
}

function roomFileMetadata(message) {
  if (!message || typeof message.id !== 'string' || !/^[A-Za-z0-9_-]{1,128}$/.test(message.id) ||
      typeof message.fileName !== 'string' || message.fileName.length < 1 || message.fileName.length > 255 ||
      /[\u0000-\u001f\u007f]/.test(message.fileName) ||
      !Number.isSafeInteger(message.fileSize) || message.fileSize < 1 || message.fileSize > ROOM_FILE_MAX_BYTES) return null;
  const fileName = message.fileName.split(/[\\/]/).pop();
  const dot = fileName.lastIndexOf('.');
  const extension = dot >= 0 ? fileName.slice(dot).toLowerCase() : '';
  if (!ROOM_FILE_TYPES.has(extension)) return { unsupported: true };
  return { id: message.id, fileName, fileSize: message.fileSize, extension };
}

function roomBytesBase64(bytes) {
  let binary = '';
  for (let index = 0; index < bytes.length; index += 0x8000) {
    binary += String.fromCharCode(...bytes.subarray(index, Math.min(index + 0x8000, bytes.length)));
  }
  return btoa(binary);
}

async function roomDownloadFile(env, metadata) {
  let response;
  try {
    response = await fetch(`https://api-data.line.me/v2/bot/message/${encodeURIComponent(metadata.id)}/content`, {
      method: 'GET',
      headers: { Authorization: `Bearer ${env.LINE_CHANNEL_ACCESS_TOKEN}` },
      signal: AbortSignal.timeout(10000)
    });
  } catch (error) {
    throw new EmployeeApiError(
      ['TimeoutError', 'AbortError'].includes(error?.name) ? 'LINE_ROOM_FILE_TIMEOUT' : 'LINE_ROOM_FILE_NETWORK_ERROR'
    );
  }
  if (!response.ok) throw new EmployeeApiError('LINE_ROOM_FILE_HTTP_ERROR', { httpStatus: response.status });
  const length = Number(response.headers.get('content-length'));
  if (Number.isFinite(length) && (length < 1 || length > ROOM_FILE_MAX_BYTES)) {
    throw new EmployeeApiError('LINE_ROOM_FILE_SIZE_INVALID');
  }
  const bytes = new Uint8Array(await response.arrayBuffer());
  if (!bytes.length || bytes.length > ROOM_FILE_MAX_BYTES) throw new EmployeeApiError('LINE_ROOM_FILE_SIZE_INVALID');
  if (metadata.extension === '.xlsx') {
    if (bytes.length < 4 || bytes[0] !== 0x50 || bytes[1] !== 0x4b || bytes[2] !== 0x03 || bytes[3] !== 0x04) {
      throw new EmployeeApiError('LINE_ROOM_FILE_CONTENT_INVALID');
    }
  } else {
    if (bytes.includes(0)) throw new EmployeeApiError('LINE_ROOM_FILE_CONTENT_INVALID');
    try { new TextDecoder('utf-8', { fatal: true }).decode(bytes); }
    catch { throw new EmployeeApiError('LINE_ROOM_FILE_CONTENT_INVALID'); }
  }
  return bytes;
}

async function handleRoomFileEvent(event, env, now) {
  const lineUserId = event.source?.type === 'user' ? event.source.userId : '';
  if (!lineUserId || !/^U[0-9a-f]{32}$/.test(lineUserId)) {
    await roomLineReply(env, event.replyToken, '為保護個人資料，請在官方帳號的一對一聊天室匯入資料。');
    return;
  }
  let access;
  try {
    access = await appsScript(env, 'roomAdmin', { operation: 'authorize', lineUserId }, now + 12000);
  } catch (error) {
    const code = logApiError('Room import authorization failed', error);
    await roomLineReply(env, event.replyToken, `${RETRY}\n參考代碼：${code}`);
    return;
  }
  if (access.status !== 'approved') {
    await roomLineReply(env, event.replyToken, ROOM_ADMIN_ONLY);
    return;
  }
  const metadata = roomFileMetadata(event.message);
  if (!metadata) {
    await roomLineReply(env, event.replyToken, event.message?.fileSize > ROOM_FILE_MAX_BYTES
      ? '檔案不可超過 3 MB。' : '檔案資訊不完整，請重新傳送。');
    return;
  }
  if (metadata.unsupported) {
    await roomLineReply(env, event.replyToken, '請傳送 .xlsx 或 .csv 檔案；舊式 .xls 請先另存成 .xlsx。');
    return;
  }
  if (!await roomLineReply(env, event.replyToken, ROOM_FILE_ACK)) return;
  try {
    const bytes = await roomDownloadFile(env, metadata);
    const result = await appsScript(env, 'roomAdmin', {
      operation: 'previewFile', lineUserId, fileName: metadata.fileName,
      fileSize: bytes.length, fileBase64: roomBytesBase64(bytes)
    }, Date.now() + 26000);
    await roomLinePush(env, lineUserId, roomImportSummary(result, 'previewFile'));
  } catch (error) {
    logApiError('Room file import failed', error);
    await roomLinePush(env, lineUserId, '無法分析這份檔案，請確認檔案格式後重新傳送。');
  }
}

async function appsScript(env, action, fields, deadline) {
  if (!env.APPS_SCRIPT_URL || !env.API_SECRET) {
    throw new EmployeeApiError('APPS_CONFIG_MISSING', {
      missingUrl: !env.APPS_SCRIPT_URL,
      missingSecret: !env.API_SECRET
    });
  }
  let url;
  try { url = new URL(env.APPS_SCRIPT_URL); }
  catch { throw new EmployeeApiError('APPS_URL_INVALID'); }
  if (url.protocol !== 'https:' || url.hostname !== 'script.google.com' ||
      !/^\/macros\/s\/[^/]+\/exec$/.test(url.pathname) || url.search || url.hash) {
    throw new EmployeeApiError('APPS_URL_INVALID');
  }
  const remaining = deadline - Date.now();
  if (remaining <= 0) throw new EmployeeApiError('APPS_TIMEOUT');
  let response;
  const timeout = action === 'roomAdmin' && fields?.operation === 'previewFile' ? 24000 : 12000;
  try {
    response = await fetch(url.toString(), {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ ...fields, action, token: env.API_SECRET }),
    // Google ContentService returns 302/303 to a one-time googleusercontent
    // URL. Fetch follows with GET; do not replay this POST.
    redirect: 'follow',
    signal: AbortSignal.timeout(Math.min(timeout, remaining))
    });
  } catch (error) {
    throw new EmployeeApiError(
      ['TimeoutError', 'AbortError'].includes(error?.name) ? 'APPS_TIMEOUT' : 'APPS_NETWORK_ERROR',
      { action }
    );
  }
  if (!response.ok) throw new EmployeeApiError('APPS_HTTP_ERROR', { action, httpStatus: response.status });
  let result;
  try { result = await response.json(); }
  catch (error) {
    throw new EmployeeApiError(
      ['TimeoutError', 'AbortError'].includes(error?.name) ? 'APPS_TIMEOUT' : 'APPS_NOT_JSON',
      { action, httpStatus: response.status }
    );
  }
  if (!result || typeof result !== 'object' || Array.isArray(result)) {
    throw new EmployeeApiError('APPS_RESPONSE_INVALID', { action });
  }
  if (result.success !== true) {
    throw new EmployeeApiError('APPS_REJECTED', {
      action,
      hasSuccessField: Object.hasOwn(result, 'success')
    });
  }
  if (typeof result.status !== 'string') {
    throw new EmployeeApiError('APPS_STATUS_MISSING', { action });
  }
  return result;
}

async function buildReply(message, lineUserId, env, deadline) {
  const diningRoute = message.match(/^dining-import:route:([A-Za-z0-9_-]{16,80}):(R01|R01-AC|R01-BD|R02)$/);
  if (diningRoute) {
    const result = await appsScript(env, 'roomAdmin', { operation: 'selectDiningRoute', lineUserId,
      importId: diningRoute[1], routeId: diningRoute[2] }, deadline);
    return roomImportSummary(result, 'selectDiningRoute');
  }
  if (message.startsWith('dining-import:')) return '桌次匯入操作無效，請重新傳送檔案。';
  const importAction = message.match(/^room-import:(confirm|cancel):([A-Za-z0-9_-]{16,80})$/);
  if (importAction) {
    const operation = importAction[1];
    const result = await appsScript(env, 'roomAdmin', { operation, lineUserId, importId: importAction[2] }, deadline);
    return roomImportSummary(result, operation);
  }
  if (/^房號批次更新(?:\s|$)/u.test(message)) {
    const result = await appsScript(env, 'roomAdmin', { operation: 'previewText', lineUserId, text: message }, deadline);
    return roomImportSummary(result, 'previewText');
  }
  if (['你好', '說明', 'help', '開始'].includes(message)) return HELP;
  const commands = { '行程': '行程', '我的行程': '行程', '車次': '車次', '我的車次': '車次', '桌次': '桌次', '房號': '房號', '旅遊攻略': '旅遊攻略', 'menu:itinerary': '行程', 'menu:vehicle': '車次', 'menu:table': '桌次', 'menu:room': '房號', 'menu:guide': '旅遊攻略' };
  const topic = commands[message];
  if (message === '綁定') return HELP;
  const input = message.replace(/^綁定[\s：:]+/u, '').trim();
  if (!topic && (!input || input.length > 50 || /[\u0000-\u001f\u007f]/.test(input))) return HELP;
  const result = await appsScript(env, 'portal', topic ? { lineUserId, topic } : { lineUserId, name: input }, deadline);
  if (result.status === 'approved') {
    const morningMessage = topic === '行程' ? r02MorningMessage(result) : null;
    const pendingMessage = topic && result.routeAssignmentPending === true ? FUSEN_ASSIGNMENT_PENDING : null;
    let diningMessage = null;
    if (topic === '桌次' && Object.hasOwn(result, 'diningReply')) {
      if (!validText(result.diningReply) || result.diningReply.length > 3800) {
        throw new EmployeeApiError('APPS_DINING_REPLY_INVALID');
      }
      diningMessage = result.diningReply;
    }
    const notices = [morningMessage, pendingMessage, diningMessage].filter(Boolean);
    let usedItineraryFallback = false;
    let imageUrls = topic === '桌次'
      ? result.imageUrls !== undefined ? result.imageUrls : result.imageUrl !== undefined ? [result.imageUrl] : undefined
      : undefined;
    if (topic === '行程') {
      imageUrls = result.imageUrls !== undefined ? result.imageUrls : result.imageUrl !== undefined ? [result.imageUrl] : undefined;
      if (imageUrls === undefined) {
        if (!validText(result.reply)) throw new EmployeeApiError('APPS_REPLY_INVALID', { action: 'trip' });
        // The Sheet links this note to the route's image URL. This also works
        // with the existing Apps Script deployment, without changing it.
        usedItineraryFallback = true;
        imageUrls = Array.from(result.reply.matchAll(/完整行程圖片[：:][ \t]*(https?:\/\/\S+)/giu), match => match[1]);
      }
      if (usedItineraryFallback && imageUrls.length === 0) {
        if (pendingMessage) return notices.length === 1 ? pendingMessage : notices;
        const missingImage = '行程圖片尚未設定，請聯絡旅遊承辦人。';
        return morningMessage ? [missingImage, morningMessage] : missingImage;
      }
    }
    if (imageUrls !== undefined) {
      if (!Array.isArray(imageUrls) || imageUrls.length < 1 || imageUrls.length > 5) {
        throw new EmployeeApiError('APPS_IMAGE_INVALID');
      }
      const messages = imageUrls.map(imageUrl => {
        const url = validImageUrl(imageUrl);
        if (!url) throw new EmployeeApiError('APPS_IMAGE_INVALID');
        return { type: 'image', originalContentUrl: url.toString(), previewImageUrl: url.toString() };
      });
      for (const notice of notices) messages.push({ type: 'text', text: notice });
      if (messages.length > 5) throw new EmployeeApiError('LINE_MESSAGES_INVALID');
      return messages.length === 1 ? messages[0] : messages;
    }
    // Only the approved, bounded itinerary text is rendered. Never render
    // employee objects, IDs, raw API messages or arbitrary response fields.
    if (!validText(result.reply)) {
      throw new EmployeeApiError('APPS_REPLY_INVALID', { action: 'trip' });
    }
    if (topic && notices.length) {
      const messages = [result.reply, ...notices.filter(notice => notice !== result.reply)];
      return messages.length === 1 ? messages[0] : messages;
    }
    return topic ? result.reply : BIND_SUCCESS;
  }
  if (['unbound', 'disabled', 'pending', 'bind_rejected'].includes(result.status) && topic) return NAME_REQUIRED;
  if (result.status === 'route_assignment_missing' && topic) return FUSEN_ASSIGNMENT_PENDING;
  if (result.status === 'busy') return '系統忙碌中，請稍後重新輸入。';
  return TRIP_UNAVAILABLE;
}

const ADMIN_ORIGIN = 'https://wsf10755005-ux.github.io';
const ADMIN_SESSION_MS = 8 * 60 * 60 * 1000;
const adminLoginAttempts = new Map();

function adminResponse(value, status = 200) {
  return new Response(status === 204 ? null : JSON.stringify(value), {
    status,
    headers: {
      'Content-Type': 'application/json; charset=utf-8', 'Cache-Control': 'no-store',
      'Access-Control-Allow-Origin': ADMIN_ORIGIN, 'Vary': 'Origin',
      'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
      'Access-Control-Allow-Headers': 'Content-Type, Authorization',
      'X-Content-Type-Options': 'nosniff'
    }
  });
}

function adminBase64(bytes) {
  return btoa(String.fromCharCode(...new Uint8Array(bytes))).replaceAll('+', '-').replaceAll('/', '_').replace(/=+$/, '');
}

function adminDecode(value) {
  const base = value.replaceAll('-', '+').replaceAll('_', '/');
  return Uint8Array.from(atob(base + '='.repeat((4 - base.length % 4) % 4)), c => c.charCodeAt(0));
}

async function adminKey(password) {
  return crypto.subtle.importKey('raw', new TextEncoder().encode('jinbin.admin.session.v1\0' + password),
    { name: 'HMAC', hash: 'SHA-256' }, false, ['sign', 'verify']);
}

async function adminSession(password, now) {
  const payload = adminBase64(new TextEncoder().encode(JSON.stringify({
    purpose: 'reminder-admin', issuedAt: now, expiresAt: now + ADMIN_SESSION_MS, nonce: crypto.randomUUID()
  })));
  const signature = await crypto.subtle.sign('HMAC', await adminKey(password), new TextEncoder().encode(payload));
  return payload + '.' + adminBase64(signature);
}

async function adminAuthorized(request, password, now) {
  const header = request.headers.get('authorization') || '';
  if (header.length > 1200 || !/^Bearer [A-Za-z0-9_-]+\.[A-Za-z0-9_-]+$/.test(header)) return false;
  try {
    const [payload, signature] = header.slice(7).split('.');
    if (!await crypto.subtle.verify('HMAC', await adminKey(password), adminDecode(signature), new TextEncoder().encode(payload))) return false;
    const session = JSON.parse(new TextDecoder().decode(adminDecode(payload)));
    return session.purpose === 'reminder-admin' && Number.isSafeInteger(session.expiresAt) &&
      Number.isSafeInteger(session.issuedAt) && session.issuedAt <= now + 30000 &&
      session.expiresAt > now && session.expiresAt - session.issuedAt === ADMIN_SESSION_MS;
  } catch { return false; }
}

async function adminPasswordMatches(candidate, expected) {
  const encode = value => new TextEncoder().encode(value);
  const [a, b] = await Promise.all([crypto.subtle.digest('SHA-256', encode(candidate)), crypto.subtle.digest('SHA-256', encode(expected))]);
  const left = new Uint8Array(a), right = new Uint8Array(b);
  let mismatch = 0;
  for (let i = 0; i < left.length; i++) mismatch |= left[i] ^ right[i];
  return mismatch === 0;
}

async function adminBody(request) {
  if (!(request.headers.get('content-type') || '').toLowerCase().startsWith('application/json')) throw new Error('invalid');
  if (Number(request.headers.get('content-length')) > 4096) throw new Error('invalid');
  if (!request.body) throw new Error('invalid');
  const reader = request.body.getReader(), chunks = [];
  let size = 0;
  try {
    while (true) {
      const part = await reader.read(); if (part.done) break;
      size += part.value.byteLength;
      if (size > 4096) { await reader.cancel(); throw new Error('invalid'); }
      chunks.push(part.value);
    }
  } finally { reader.releaseLock(); }
  const bytes = new Uint8Array(size); let offset = 0;
  for (const chunk of chunks) { bytes.set(chunk, offset); offset += chunk.byteLength; }
  const text = new TextDecoder().decode(bytes);
  const body = JSON.parse(text);
  if (!body || typeof body !== 'object' || Array.isArray(body)) throw new Error('invalid');
  return body;
}

async function handleAdminRequest(request, env) {
  const origin = request.headers.get('origin');
  if (origin && origin !== ADMIN_ORIGIN) return adminResponse({ error: '不允許的來源。' }, 403);
  if (request.method === 'OPTIONS') return adminResponse(null, 204);
  const path = new URL(request.url).pathname;
  if (path === '/admin/health' && request.method === 'GET') {
    return adminResponse({ service: 'employee-trip', build: 'workbook-vehicle-20261113',
      adminConfigured: typeof env.ADMIN_PASSWORD === 'string' && env.ADMIN_PASSWORD.trim().length >= 12 });
  }
  if (!['/admin/login', '/admin/reminders'].includes(path)) return adminResponse({ error: '找不到此功能。' }, 404);
  const password = env.ADMIN_PASSWORD;
  if (typeof password !== 'string' || password.trim().length < 12) {
    return adminResponse({ error: '管理功能尚未啟用，請先完成管理密碼設定。', code: 'ADMIN_NOT_CONFIGURED' }, 503);
  }
  const now = Date.now();
  if (path === '/admin/login') {
    if (request.method !== 'POST') return adminResponse({ error: '不支援此操作。' }, 405);
    for (const [ip, attempt] of adminLoginAttempts) if (attempt.until <= now) adminLoginAttempts.delete(ip);
    const ip = (request.headers.get('cf-connecting-ip') || 'unknown').slice(0, 64);
    const attempt = adminLoginAttempts.get(ip) || { count: 0, until: now + 15 * 60000 };
    if (attempt.count >= 5) return adminResponse({ error: '登入嘗試次數過多，請稍後再試。' }, 429);
    if (adminLoginAttempts.size >= 2000) adminLoginAttempts.delete(adminLoginAttempts.keys().next().value);
    attempt.count++;
    adminLoginAttempts.set(ip, attempt);
    let body;
    try { body = await adminBody(request); } catch { return adminResponse({ error: '輸入格式不正確。' }, 400); }
    if (typeof body.password !== 'string' || body.password.length > 512 || !await adminPasswordMatches(body.password, password)) {
      return adminResponse({ error: '管理密碼不正確。' }, 401);
    }
    adminLoginAttempts.delete(ip);
    return adminResponse({ token: await adminSession(password, now), expiresAt: now + ADMIN_SESSION_MS });
  }
  if (!await adminAuthorized(request, password, now)) return adminResponse({ error: '請重新登入。' }, 401);
  if (!['GET', 'POST'].includes(request.method)) return adminResponse({ error: '不支援此操作。' }, 405);
  let fields = { operation: 'list' };
  if (request.method === 'POST') {
    let body;
    try { body = await adminBody(request); } catch { return adminResponse({ error: '輸入格式不正確。' }, 400); }
    if (!['create', 'update', 'cancel'].includes(body.operation)) return adminResponse({ error: '不支援此操作。' }, 400);
    fields = { operation: body.operation };
    for (const key of ['submissionId', 'id', 'rowNumber', 'version', 'routeIds', 'date', 'time', 'placeName', 'mapUrl']) {
      if (Object.hasOwn(body, key)) fields[key] = body[key];
    }
  }
  try {
    const result = await appsScript(env, 'adminReminders', fields, now + 18000);
    if (result.status === 'ok') return adminResponse(result);
    const messages = {
      invalid_request: '輸入資料不完整，請重新檢查。', invalid_time: '日期或時間不正確。',
      too_soon: '集合時間需距現在至少 11 分鐘。', invalid_place: '請填寫集合地點，或可直接開啟的 HTTPS 地圖連結；地點加網址最多 200 字。',
      invalid_route: '找不到選擇的路線，請重新整理。', conflict: '這筆資料已變更，請重新整理後再試。',
      duplicate: '這個路線與時間已有提醒，請修改原本那一筆。', locked: '這筆已取消、已發送或已進入發送時段，無法再修改。',
      not_found: '找不到這筆提醒。', busy: '目前正在更新提醒，請稍後再試。'
    };
    return adminResponse({ error: messages[result.status] || '暫時無法處理，請稍後再試。', code: result.status },
      ['conflict', 'duplicate', 'locked'].includes(result.status) ? 409 : result.status === 'busy' ? 503 : 400);
  } catch (error) {
    logApiError('Admin reminder request failed', error);
    return adminResponse({ error: '管理頁尚未連接完成，請確認後台程式已更新。', code: 'ADMIN_BACKEND_UNAVAILABLE' }, 502);
  }
}

function validReminderBatch(batch) {
  const id = value => typeof value === 'string' && value.length > 0 && value.length <= 200 && !/[\u0000-\u001f\u007f]/.test(value);
  return batch && typeof batch === 'object' && !Array.isArray(batch) &&
    id(batch.reminderId) && id(batch.batchId) &&
    typeof batch.retryKey === 'string' && /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(batch.retryKey) &&
    Array.isArray(batch.to) && batch.to.length > 0 && batch.to.length <= 100 &&
    batch.to.every(value => typeof value === 'string' && /^U[0-9a-f]{32}$/.test(value)) &&
    new Set(batch.to).size === batch.to.length && validText(batch.text);
}

async function deliverReminders(env) {
  if (!env.LINE_CHANNEL_ACCESS_TOKEN) throw new EmployeeApiError('LINE_CONFIG_MISSING');
  const deadline = Date.now() + 28000;
  const result = await appsScript(env, 'claimReminders', {}, deadline);
  if (result.status !== 'ok' || !Array.isArray(result.batches)) {
    throw new EmployeeApiError('APPS_REMINDERS_INVALID', { action: 'claimReminders' });
  }
  if (result.batches.length === 0) return;
  if (!result.batches.every(validReminderBatch)) throw new EmployeeApiError('APPS_REMINDER_BATCH_INVALID');
  const needed = result.batches.reduce((total, batch) => total + batch.to.length, 0);
  await ensureReminderQuota(env, needed, deadline);
  // A failed batch must not prevent independent routes from being notified.
  // Backend leases retain each payload and retry key until acknowledgement.
  const outcomes = await Promise.allSettled(result.batches.map(batch => deliverReminderBatch(env, batch, deadline)));
  for (const outcome of outcomes) {
    if (outcome.status === 'rejected') logApiError('Reminder batch failed', outcome.reason);
  }
}

async function lineQuotaJson(env, path, deadline) {
  const remaining = deadline - Date.now();
  if (remaining <= 0) throw new EmployeeApiError('LINE_QUOTA_TIMEOUT');
  let response, result;
  try {
    response = await fetch(`https://api.line.me/v2/bot/message/${path}`, {
      method: 'GET',
      headers: { Authorization: `Bearer ${env.LINE_CHANNEL_ACCESS_TOKEN}` },
      signal: AbortSignal.timeout(Math.min(4000, remaining))
    });
  } catch (error) {
    throw new EmployeeApiError(
      ['TimeoutError', 'AbortError'].includes(error?.name) ? 'LINE_QUOTA_TIMEOUT' : 'LINE_QUOTA_NETWORK_ERROR'
    );
  }
  if (!response.ok) throw new EmployeeApiError('LINE_QUOTA_HTTP_ERROR', { httpStatus: response.status });
  try { result = await response.json(); }
  catch (error) {
    throw new EmployeeApiError(
      ['TimeoutError', 'AbortError'].includes(error?.name) ? 'LINE_QUOTA_TIMEOUT' : 'LINE_QUOTA_RESPONSE_INVALID'
    );
  }
  if (!result || typeof result !== 'object' || Array.isArray(result)) throw new EmployeeApiError('LINE_QUOTA_RESPONSE_INVALID');
  return result;
}

async function ensureReminderQuota(env, needed, deadline) {
  const quota = await lineQuotaJson(env, 'quota', deadline);
  if (quota.type === 'none') return;
  if (quota.type !== 'limited' || !Number.isSafeInteger(quota.value) || quota.value < 0) {
    throw new EmployeeApiError('LINE_QUOTA_RESPONSE_INVALID');
  }
  const consumption = await lineQuotaJson(env, 'quota/consumption', deadline);
  if (!Number.isSafeInteger(consumption.totalUsage) || consumption.totalUsage < 0) {
    throw new EmployeeApiError('LINE_QUOTA_RESPONSE_INVALID');
  }
  const remaining = Math.max(0, quota.value - consumption.totalUsage);
  if (remaining < needed) throw new EmployeeApiError('LINE_QUOTA_INSUFFICIENT', { needed, remaining });
}

async function deliverReminderBatch(env, batch, deadline) {
  if (!validReminderBatch(batch)) throw new EmployeeApiError('APPS_REMINDER_BATCH_INVALID');
  const remaining = deadline - Date.now();
  if (remaining <= 0) throw new EmployeeApiError('LINE_REMINDER_TIMEOUT');
  let response;
  try {
    response = await fetch('https://api.line.me/v2/bot/message/multicast', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        Authorization: `Bearer ${env.LINE_CHANNEL_ACCESS_TOKEN}`,
        'X-Line-Retry-Key': batch.retryKey
      },
      body: JSON.stringify({ to: batch.to, messages: [{ type: 'text', text: batch.text }] }),
      signal: AbortSignal.timeout(Math.min(8000, remaining))
    });
  } catch (error) {
    throw new EmployeeApiError(
      ['TimeoutError', 'AbortError'].includes(error?.name) ? 'LINE_REMINDER_TIMEOUT' : 'LINE_REMINDER_NETWORK_ERROR'
    );
  }
  const acceptedRetry = response.status === 409 && Boolean(response.headers.get('x-line-accepted-request-id'));
  if (!response.ok && !acceptedRetry) {
    throw new EmployeeApiError('LINE_REMINDER_HTTP_ERROR', { httpStatus: response.status });
  }
  const ack = await appsScript(env, 'ackReminder', {
    reminderId: batch.reminderId,
    batchId: batch.batchId,
    retryKey: batch.retryKey
  }, deadline);
  if (ack.status !== 'ok') throw new EmployeeApiError('APPS_REMINDER_ACK_INVALID', { action: 'ackReminder' });
}
